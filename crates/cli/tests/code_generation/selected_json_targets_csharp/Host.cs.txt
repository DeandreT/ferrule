    private static void Write(string directory, string leaf, byte[] body)
    {
        using var file = new FileStream(Path.Combine(directory, leaf), FileMode.CreateNew, FileAccess.Write, FileShare.Read);
        file.Write(body); file.Flush(true);
    }

    private static byte[] Utf16(string text)
    {
        var bytes = new byte[checked(text.Length * 2)];
        for (var index = 0; index < text.Length; index++)
        { bytes[index * 2] = (byte)text[index]; bytes[index * 2 + 1] = (byte)(text[index] >> 8); }
        return bytes;
    }
    private static byte[] RecipeBytes(JsonObject recipe)
    {
        byte[] result;
        if (recipe["utf8_hex"] is JsonValue hex) result = Convert.FromHexString(hex.GetValue<string>());
        else if (recipe["complete_byte_recipe"] is JsonObject full) result = RecipeBytes(full);
        else if (recipe["concatenate"] is JsonArray parts)
        {
            using var stream = new MemoryStream();
            foreach (var part in parts) stream.Write(RecipeBytes(part!.AsObject()));
            result = stream.ToArray();
        }
        else if (recipe["literal_utf8"] is JsonValue literal) result = StrictUtf8.GetBytes(literal.GetValue<string>());
        else if (recipe["repeat_utf8"] is JsonValue repeated)
        {
            var unit = StrictUtf8.GetBytes(repeated.GetValue<string>());
            var count = recipe["count"]!.GetValue<int>();
            result = new byte[checked(unit.Length * count)];
            for (var index = 0; index < count; index++) unit.CopyTo(result, index * unit.Length);
        }
        else throw new InvalidOperationException("Unknown complete byte recipe.");
        var length = recipe["exact_bytes"] ?? recipe["bytes"];
        if (length is not null && result.Length != length.GetValue<int>())
            throw new InvalidOperationException("Independent byte recipe length differs.");
        if (recipe["text"] is JsonValue text && !StrictUtf8.GetBytes(text.GetValue<string>()).AsSpan().SequenceEqual(result))
            throw new InvalidOperationException("Independent text/hex recipe bodies differ.");
        if (recipe["sha256"] is JsonValue digest && !string.Equals(Convert.ToHexString(SHA256.HashData(result)), digest.GetValue<string>(), StringComparison.OrdinalIgnoreCase))
            throw new InvalidOperationException("Independent byte recipe digest differs.");
        return result;
    }
    private static string RecipeString(JsonNode value)
    {
        if (value is JsonValue text) return text.GetValue<string>();
        var recipe = value["complete_value_recipe"]!.AsObject();
        var scalar = recipe["repeat_unicode_scalar"]!.GetValue<string>();
        var builder = new StringBuilder(checked(scalar.Length * recipe["count"]!.GetValue<int>()));
        for (var index = 0; index < recipe["count"]!.GetValue<int>(); index++) builder.Append(scalar);
        return builder.ToString();
    }
    private static string InputText(JsonObject recipe) => recipe["utf16_code_units"] is JsonArray units
        ? new string(units.Select(unit => (char)unit!.GetValue<int>()).ToArray())
        : StrictUtf8.GetString(RecipeBytes(recipe));
    private static FerruleValue Scalar(JsonObject spec)
    {
        if (spec["String"] is JsonNode text) return FerruleValue.FromString(RecipeString(text));
        if (spec["Int"] is JsonNode integer) return FerruleValue.FromInt64(integer.GetValue<long>());
        if (spec["Bool"] is JsonNode boolean) return FerruleValue.FromBoolean(boolean.GetValue<bool>());
        throw new InvalidOperationException("Unknown independent expected scalar.");
    }
    private static FerruleInstance Instance(JsonObject spec)
    {
        if (spec["Scalar"] is JsonObject scalar) return new FerruleScalar(Scalar(scalar));
        if (spec["Group"] is JsonObject group)
        {
            if (group["origin"]!.AsObject().Count != 1 || !group["origin"]!.AsObject().ContainsKey("Unknown"))
                throw new InvalidOperationException("This frozen JSON boundary expects Unknown origins.");
            return new FerruleGroup(group["ordered_fields"]!.AsArray().Select(field => field!.AsArray())
                .Select(field => new FerruleField(field[0]!.GetValue<string>(), Instance(field[1]!.AsObject()))));
        }
        if (spec["Repeated"] is JsonArray repeated) return new FerruleRepeated(repeated.Select(item => Instance(item!.AsObject())));
        throw new InvalidOperationException("Unknown independent expected instance.");
    }
    private static object Boundary(FerruleFilterMapBoundary at) => new {

    private sealed class Cancellation(string specification) : IFerruleFilterMapCancellation
    {
        internal readonly List<FerruleFilterMapBoundary> Calls = new();
        public bool IsCancelled(FerruleFilterMapBoundary at)
        {
            Calls.Add(at);
            return specification switch {
                "true for every consulted boundary" => true,
                "true exactly at SourceReservation; false at every other boundary" => at.Kind == FerruleFilterMapBoundaryKind.SourceReservation,
                _ => throw new InvalidOperationException("Unknown frozen cancellation recipe."),
            };
        }
    }
    private sealed class Loader(JsonObject specification, string directory, string prefix) : IFerruleDynamicJsonSourceLoader
    {
        internal readonly List<object> Calls = new();
        internal readonly List<Exception> Errors = new();
        public byte[] Load(string sourceName, string logicalPath)
        {
            var index = Calls.Count;
            var request = new { source = sourceName, path = logicalPath };
            Calls.Add(request); Json(directory, $"{prefix}-loader{index}-request.original.json", request);
            var responses = specification["ordered_responses"]!.AsArray();
            var response = index < responses.Count ? responses[index]!.AsObject() : null;
            if (response is null || response["source"]!.GetValue<string>() != sourceName || response["path"]!.GetValue<string>() != logicalPath)
            {
                var error = new InvalidOperationException("218 unexpected loader request");
                Errors.Add(error); Capture.Retain(directory, $"{prefix}-loader{index}-error", error); throw error;
            }
            if (response["result"]!["Ok"] is JsonObject okay)
            {
                var bytes = RecipeBytes(okay);
                Write(directory, $"{prefix}-loader{index}-reply.original.bytes", bytes);
                return bytes;
            }
            var refusal = new InvalidOperationException(response["result"]!["Err"]!.GetValue<string>());
            Errors.Add(refusal); Capture.Retain(directory, $"{prefix}-loader{index}-error", refusal); throw refusal;
        }
    }

    private static FerruleRuntimeException RuntimeExpected(JsonObject cause)
    {
        var fields = cause["fields"]!.AsObject();
        var kind = cause["kind"]!.GetValue<string>();
        var name = fields["name"]?.GetValue<string>();
        var message = fields["message"]?.GetValue<string>();
        var source = fields["source"]?.GetValue<string>();
        var path = fields["path"]?.GetValue<string>();
        return kind switch {
            "UnknownTarget" => new(FerruleRuntimeError.UnknownTarget, fields["display"]!.GetValue<string>(), detail: name),
            "MissingNamedSource" => new(FerruleRuntimeError.MissingNamedSource, $"named source {name} is required by this mapping", detail: name),
            "DuplicateNamedSource" => new(FerruleRuntimeError.DuplicateNamedSource, $"named source '{name}' was supplied more than once", detail: name),
            "UnexpectedNamedSource" => new(FerruleRuntimeError.UnexpectedNamedSource, $"named source '{name}' is not declared by this mapping", detail: name),
            "MissingRuntimeParameter" => new(FerruleRuntimeError.MissingRuntimeParameter,
                $"node {fields["node"]}: execution context does not provide runtime parameter '{name}'", node: fields["node"]!.GetValue<uint>(), runtimeParameter: name),
            "MappingException" => new(FerruleRuntimeError.MappingException, $"node {fields["node"]}: mapping exception: {message}", node: fields["node"]!.GetValue<uint>(), mappingExceptionMessage: message),
            "MappingFailure" => new(FerruleRuntimeError.MappingFailure, $"mapping failure rule {fields["rule"]}: {message}", failureRule: fields["rule"]!.GetValue<int>(), mappingFailureMessage: message),
            "MissingDynamicSourceLoader" => new(FerruleRuntimeError.MissingDynamicSourceLoader, $"dynamic source '{source}' requires a host source loader", sourceField: source),
            "DynamicSourceLoad" => new(FerruleRuntimeError.DynamicSourceLoad, $"dynamic source '{source}' could not load '{path}': {message}", new InvalidOperationException(message), detail: path, sourceField: source),
            _ => throw new InvalidOperationException("Unknown frozen runtime error."),
        };
    }
    private static JsonNode FilterExpected(JsonObject cause)
    {
        var fields = cause["fields"]!.AsObject(); var boundary = Boundary(fields["boundary"]!.AsObject());
        var source = fields["source"]!.AsObject(); var budget = source["kind"]!.GetValue<string>() == "FilterMapBudget";
        var amounts = source["fields"]!.AsObject(); var kind = budget ? "Budget" : "Cancelled";
        var budgetKind = budget ? amounts["kind"]!.GetValue<string>() : null;
        var used = budget ? amounts["used"]!.ToJsonString() : null;
        var requested = budget ? amounts["requested"]!.ToJsonString() : null;
        var maximum = budget ? amounts["max"]!.ToJsonString() : null;
        var message = $"filter/map {kind}: expected=; found=; bits=; budget={budgetKind}; used={used}; requested={requested}; maximum={maximum}";
        var inner = Capture.Json(Capture.Error(new Exception(message)))!.AsObject();
        inner["type"] = "Ferrule.Runtime.FerruleFilterMapFailure";
        inner["feature"] = Capture.Json(new { kind, expected = (string?)null, found = (object?)null, bits = (string?)null,
            budget = budgetKind, used, requested, maximum });
        var outer = Capture.Json(Capture.Error(new Exception($"filter/map {boundary}: {message}")))!.AsObject();
        outer["type"] = "Ferrule.Runtime.FerruleFilterMapException"; outer["inner"] = inner;
        outer["envelope"] = Capture.Json(new { item = boundary.Item, phase = boundary.Phase.ToString(), captureIndex = boundary.CaptureIndex,
            sourcePosition = boundary.SourcePosition, function = boundary.Function, node = boundary.Node, kind = boundary.Kind.ToString() });
        return outer;
    }
    private static bool ErrorMatches(Exception actual, JsonObject expected)
    {
        var kind = expected["kind"]!.GetValue<string>(); var fields = expected["fields"]!.AsObject();
        if (kind != "Execution")
        {
            if (actual is not FerruleRuntimeException boundary || boundary.Error != FerruleRuntimeError.JsonBoundary) return false;
            var wanted = new FerruleRuntimeException(FerruleRuntimeError.JsonBoundary, fields["csharp_exact_message"]!.GetValue<string>(), detail: fields["csharp_exact_message"]!.GetValue<string>());
            var fieldsExact = JsonNode.DeepEquals(Capture.Json(Capture.Error(boundary))!["runtime"], Capture.Json(Capture.Error(wanted))!["runtime"]);
            var innerType = fields["csharp_inner_type"]?.GetValue<string>();
            return fieldsExact && boundary.Message == wanted.Message && boundary.InnerException?.GetType().FullName == innerType;
        }
        var cause = fields["cause"]!.AsObject(); var causeKind = cause["kind"]!.GetValue<string>();
        if (causeKind == "FilterMapRuntime") return JsonNode.DeepEquals(Capture.Json(Capture.Error(actual)), FilterExpected(cause));
        if (causeKind == "DynamicSourceLoad" && cause["fields"]!["csharp_captured_decoder_message"] is JsonValue)
        {
            var details = cause["fields"]!.AsObject();
            var wanted = new FerruleRuntimeException(FerruleRuntimeError.DynamicSourceLoad, "unused framework-dependent display",
                detail: details["path"]!.GetValue<string>(), sourceField: details["source"]!.GetValue<string>());
            return actual is FerruleRuntimeException found && found.InnerException is DecoderFallbackException &&
                JsonNode.DeepEquals(Capture.Json(Capture.Error(found))!["runtime"], Capture.Json(Capture.Error(wanted))!["runtime"]) &&
                found.Message.StartsWith($"dynamic source '{details["source"]!.GetValue<string>()}' could not load '{details["path"]!.GetValue<string>()}': ", StringComparison.Ordinal);
        }
        return JsonNode.DeepEquals(Capture.Json(Capture.Error(actual)), Capture.Json(Capture.Error(RuntimeExpected(cause))));
    }
    private static FerruleExecutionContext? Context(JsonObject control, Cancellation? checker, List<object> parameterOriginals)
    {
        var host = control["execution_context"] as JsonObject;
        var controls = control["filter_map_controls"] as JsonObject;
        if (host is null && controls is null) return null;
        var parameters = host is null ? FerruleRuntimeParameters.Empty : new FerruleRuntimeParameters(
            host["parameters"]!.AsArray().Select(row => row!.AsObject()).Select(row =>
                MakeParameter(row, parameterOriginals)));

        var limits = controls is null ? FerruleFilterMapLimits.Default : new FerruleFilterMapLimits(
            UInt128.Parse(controls["source_items"]!.ToJsonString(), CultureInfo.InvariantCulture),
            UInt128.Parse(controls["work"]!.ToJsonString(), CultureInfo.InvariantCulture));
        return new(host?["mapping_file_path"]?.GetValue<string>() ?? "logical-218.json",
            host?["main_mapping_file_path"]?.GetValue<string>() ?? "logical-218.json",
            host?["current_datetime"]?.GetValue<string>(), parameters, limits, checker);
    }
    private static KeyValuePair<string, FerruleValue> MakeParameter(JsonObject row, List<object> originals)
    {
        var name = row["name"]!.GetValue<string>(); var value = Scalar(row["value"]!.AsObject());
        originals.Add(new { name, value = Capture.Value(value) });
        return new(name, value);
    }

            }
            var actual = Capture.Attempt(() => byteApi
                ? InvokeBytes(profile, primaryBytes!, selected, inputs.ToArray(), context, loader)
                : InvokeText(profile, primaryText!, selected, inputs.ToArray(), context, loader));
            // Retain every actual public result/error and callback before any status/equality comparison.
            Capture.Retain(args[3], prefix + "-COMPLETE-ACTUAL-PUBLIC-RESULT", actual);
            Json(args[3], prefix + "-COMPLETE-LOADER-CALLS-ORIGINAL.json", loader?.Calls ?? new List<object>());
            Json(args[3], prefix + "-COMPLETE-CANCELLATION-ORIGINAL.json", checker?.Calls.Select(Boundary).ToArray() ?? Array.Empty<object>());
            var document = Extract(actual); byte[]? actualBytes = null; object? decoded = null;
            if (document is not null)
            {
                Json(args[3], prefix + "-ACTUAL-VARIANT-NAME-ORIGINAL.json", new { document.Variant, document.Name });
                if (document.Text is not null) Write(args[3], prefix + "-ACTUAL-DOCUMENT.original.utf16le", Utf16(document.Text));
                var conversion = Capture.Attempt(() => (object)(document.Bytes ?? StrictUtf8.GetBytes(document.Text!)));
                Capture.Retain(args[3], prefix + "-COMPLETE-ACTUAL-WIRE-CONVERSION", conversion);
                if (conversion is byte[] bytes)
                {
                    actualBytes = bytes;
                    Write(args[3], prefix + "-ACTUAL-DOCUMENT.original.bytes", actualBytes);
                    // This is an observation of the returned JSON wire, never a second mapping execution.
                    decoded = Capture.Attempt(() => {
                        var descriptor = document.Variant == "Primary" ? schemas[profileId]!["primary"]!.GetValue<string>()
                            : schemas[profileId]!["named"]![document.Name!]!.GetValue<string>();
                        return FerruleJson.ParseEmbeddedBytes(descriptor, bytes);
                    });
                    Capture.Retain(args[3], prefix + "-COMPLETE-ACTUAL-DECODED-WIRE-INSTANCE", decoded);
                }
            }
            bool matched;
            if (expected["status"]!.GetValue<string>() == "Ok")
                matched = document is not null && document.Variant == expected["variant"]!.GetValue<string>() &&
                    document.Name == expected["name"]?.GetValue<string>() && actualBytes is not null && actualBytes.AsSpan().SequenceEqual(wantedBytes) &&
                    decoded is FerruleInstance actualInstance && JsonNode.DeepEquals(Capture.Json(Capture.Tree(actualInstance)), Capture.Json(Capture.Tree(wantedInstance!)));
            else matched = actual is Exception error && document is null && ErrorMatches(error, expected);
            matched &= JsonNode.DeepEquals(Capture.Json(loader?.Calls ?? new List<object>()), control["expected_loader_calls"]);
            matched &= JsonNode.DeepEquals(Capture.Json(checker?.Calls.Select(Boundary).ToArray() ?? Array.Empty<object>()), control["expected_filter_map_cancellation_calls"]);
            if (control["id"]!.GetValue<string>() == "selected-dynamic-host-refusal")
                matched &= actual is FerruleRuntimeException refusal && loader is not null && loader.Errors.Count == 1 && ReferenceEquals(refusal.InnerException, loader.Errors[0]);
            allMatched &= matched;
            var check = new { id = control["id"]!.GetValue<string>(), api, matched, actualType = actual.GetType().FullName,
                observedWireOnly = true, rawMappingInstanceIsNotPubliclyReturned = true };
            checks.Add(check); Json(args[3], prefix + "-COMPARISON.json", check);
            Console.WriteLine($"selected-json218 {count:00} {api} matched={matched}");
        }
        Json(args[3], "COMPLETE-COHORT-OUTCOMES.json", checks);
        var wantedCount = args[2] == "small" ? 65 : 12;
        return allMatched && count == wantedCount ? 0 : 1;
    }
