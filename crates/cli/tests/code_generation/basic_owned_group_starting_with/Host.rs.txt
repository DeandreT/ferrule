use codegen_runtime::{AggregateFunction,BasicNamedSourceInput,ExecutionContext,Instance,InstanceKind,OwnedSourceError,
    OwnedSourceIdentity,OwnedSourceLimits,RuntimeError,SourcePathError,TargetSelection};
use serde_json::{Value as Json,json};
use std::{fs,io::Write as _,path::{Path,PathBuf}};
mod codec;
mod legacy_control;
mod proof;
type BorrowedOutcome<'a>=Result<Vec<(Option<&'a str>,&'a Instance)>,&'a RuntimeError>;
type InputSnapshot=(Json,Vec<Json>);
fn write(directory:&Path,name:&str,bytes:&[u8])->std::io::Result<()> {
    let mut file=fs::OpenOptions::new().create_new(true).write(true).open(directory.join(name))?;file.write_all(bytes)?;file.flush()
}
fn raw_inputs(source:&Instance,named:&[(String,Instance)])->Json {
    json!({"source":codec::raw_tree(source),"inputs":named.iter().map(|(name,value)|json!([name,codec::raw_tree(value)])).collect::<Vec<_>>()})
}
fn strings(recipe:&Json,key:&str)->Vec<String> {recipe[key].as_array().unwrap().iter().map(|s|s.as_str().unwrap().into()).collect()}
fn aggregate_function(recipe:&Json)->AggregateFunction {
    match recipe["function"].as_str().unwrap() {"Count"=>AggregateFunction::Count,"Sum"=>AggregateFunction::Sum,"Avg"=>AggregateFunction::Avg,
        "Min"=>AggregateFunction::Min,"Max"=>AggregateFunction::Max,"Join"=>AggregateFunction::Join,"ItemAt"=>AggregateFunction::ItemAt,_=>panic!("authored aggregate function")}
}
fn error(recipe:&Json)->RuntimeError {
    let node=recipe["node"].as_u64().map(|n|n as u32);
    let found=match recipe["found"].as_str() {Some("bool")=>"bool",Some("string")=>"string",Some("json null")=>"json null",Some("int")=>"int",Some("float")=>"float",Some("xml nil")=>"xml nil",_=>"null"};
    match recipe["kind"].as_str().unwrap() {
        "FunctionType"|"FunctionInvalidArgument"=>codec::function_error(recipe).into(),
        "GeneratedSequenceTooLarge"=>RuntimeError::GeneratedSequenceTooLarge {requested:recipe["requested"].as_str().unwrap().parse().unwrap(),max:recipe["maximum"].as_str().unwrap().parse().unwrap()},
        "InvalidTokenizeRegexFlags"=>RuntimeError::InvalidTokenizeRegexFlags {flags:recipe["flags"].as_str().unwrap().into()},
        "ZeroWidthTokenizeRegex"=>RuntimeError::ZeroWidthTokenizeRegex,
        "AggregateIntegerOverflow"=>RuntimeError::AggregateIntegerOverflow {function:aggregate_function(recipe)},
        "AggregateNonFinite"=>RuntimeError::AggregateNonFinite {function:aggregate_function(recipe)},
        "MappingException"=>RuntimeError::MappingException {node:node.unwrap(),message:recipe["message"].as_str().map(str::to_owned)},
        "MappingFailure"=>RuntimeError::MappingFailure {rule:recipe["rule"].as_u64().unwrap() as usize,message:recipe["message"].as_str().map(str::to_owned)},
        "NotABool"=>RuntimeError::NotABool {node:node.unwrap(),found},"NotAnItemCount"=>RuntimeError::NotAnItemCount {node:node.unwrap(),found},
        "InvalidBlockSize"=>RuntimeError::InvalidBlockSize {node:node.unwrap()},
        "UnknownTarget"=>RuntimeError::UnknownTarget {name:recipe["name"].as_str().unwrap().into()},
        "OwnedSource"=>{
            let owner=if recipe["owner"]=="Primary" {OwnedSourceIdentity::Primary} else {OwnedSourceIdentity::Named(recipe["owner"].as_str().unwrap().into())};
            let source=match recipe["cause"].as_str().unwrap() {
                "SourceUnavailable"=>OwnedSourceError::SourceUnavailable {name:recipe["owner"].as_str().unwrap().into()},
                "InactiveSourceFrame"=>OwnedSourceError::InactiveSourceFrame {owner,collection:Some(strings(recipe,"collection"))},
                "MissingDocumentPath"=>OwnedSourceError::MissingDocumentPath {owner},
                "UniteratedRepetition"=>OwnedSourceError::UniteratedRepetition {owner,path:strings(recipe,"path"),segment:recipe["segment"].as_u64().unwrap() as usize},
                "CannotTraverse"=>OwnedSourceError::Traversal {owner,source:SourcePathError::CannotTraverse {path:strings(recipe,"path"),segment:recipe["segment"].as_u64().unwrap() as usize,found:InstanceKind::Scalar}},
                "ExpectedScalar"=>OwnedSourceError::Traversal {owner,source:SourcePathError::ExpectedScalar {path:strings(recipe,"path"),found:InstanceKind::Group}},
                _=>panic!("authored generated cause"),
            };if let Some(node)=node {RuntimeError::OwnedSourceAtNode {node,source}} else {RuntimeError::OwnedSource(source)}
        },_=>panic!("authored runtime error"),
    }
}
fn observe(actual:BorrowedOutcome<'_>,call:&Json,before:&InputSnapshot,source:&Instance,named:&[(String,Instance)])->Json {
    let trees=actual.as_ref().ok().map(|outputs|json!(outputs.iter().map(|(name,tree)|json!([name,codec::encode(tree)])).collect::<Vec<_>>()));
    let after=(codec::encode(source),named.iter().map(|(name,tree)|json!([name,codec::encode(tree)])).collect::<Vec<_>>());
    let actual_error=actual.as_ref().err().map(|error|json!({"debug":format!("{error:#?}"),"display":error.to_string()}));
    let matched=match actual {
        Ok(_)=>call["expected"].get("Ok").is_some_and(|expected|trees.as_ref().is_some_and(|actual|actual==expected)),
        Err(actual)=>call["expected"].get("Err").is_some_and(|recipe|*actual==error(recipe)),
    }&&*before==after;
    json!({"actual_trees":trees,"actual_error":actual_error,"after_inputs":after,"matched":matched})
}
macro_rules! invoke {
    ($module:ident,$selection:expr,$source:expr,$inputs:expr,$context:expr,$limits:expr,$directory:expr,$prefix:expr,$call:expr,$before:expr,$named:expr,$identity:expr,$raw_before:expr)=>{{
        if $selection=="All" {
            let invocation=std::panic::catch_unwind(std::panic::AssertUnwindSafe(||
                $module::execute_with_basic_owned_sources($source,$inputs,Some($context),$limits)));
            codec::retain_outcome_with_input_after(invocation,
                |suffix,bytes|proof::write_call($directory,$prefix,$identity,suffix,bytes),
                |output|json!({"variant":"All","primary":codec::raw_tree(&output.primary),
                    "extras":output.extras.iter().map(|output|json!({"name":output.name,"instance":codec::raw_tree(&output.instance)})).collect::<Vec<_>>()}),
                ||{let state=raw_inputs($source,$named);json!({"unchanged":state==*$raw_before,"state":state})},
                |product|{
                    let actual=match product {
                        Ok(output)=>{let mut result=vec![(None,&output.primary)];result.extend(output.extras.iter().map(|output|(Some(output.name.as_str()),&output.instance)));Ok(result)},
                        Err(error)=>Err(error),
                    };
                    observe(actual,$call,$before,$source,$named)
                })
        } else {let target=if $selection=="Primary" {TargetSelection::Primary} else {TargetSelection::Named($selection)};
            let invocation=std::panic::catch_unwind(std::panic::AssertUnwindSafe(||
                $module::execute_selected_with_basic_owned_sources($source,$inputs,Some($context),$limits,target)));
            codec::retain_outcome_with_input_after(invocation,
                |suffix,bytes|proof::write_call($directory,$prefix,$identity,suffix,bytes),
                |output|match output {
                    $module::SelectedTargetOutput::Primary(instance)=>json!({"variant":"Primary","instance":codec::raw_tree(instance)}),
                    $module::SelectedTargetOutput::Named(output)=>json!({"variant":"Named","name":output.name,"instance":codec::raw_tree(&output.instance)}),
                },
                ||{let state=raw_inputs($source,$named);json!({"unchanged":state==*$raw_before,"state":state})},
                |product|{
                    let actual=match product {
                        Ok($module::SelectedTargetOutput::Primary(instance))=>Ok(vec![(None,instance)]),
                        Ok($module::SelectedTargetOutput::Named(output))=>Ok(vec![(Some(output.name.as_str()),&output.instance)]),
                        Err(error)=>Err(error),
                    };
                    observe(actual,$call,$before,$source,$named)
                })
        }
    }};
}
fn main()->Result<(),Box<dyn std::error::Error>> {
    let args=std::env::args_os().skip(1).collect::<Vec<_>>();assert_eq!(args.len(),3);
    let directory=PathBuf::from(&args[1]);assert!(directory.is_absolute()&&!directory.exists());fs::create_dir(&directory)?;
    let contract=proof::begin(&directory,Path::new(&args[2]))?;
    let raw=fs::read(&args[0])?;write(&directory,"COMPLETE-FROZEN-CONTROLS.json",&raw)?;let controls:Json=serde_json::from_slice(&raw)?;
    let limits=OwnedSourceLimits {max_roots:16,max_active_frames:32,max_path_segments:64,max_metadata_bytes:16_777_216,max_walk_contexts:512};
    let execution=ExecutionContext::new(Path::new("itemat473.mapping"));let mut checks=Vec::new();
    for (ordinal,case) in controls["cases"].as_array().unwrap().iter().enumerate() {
        let source=codec::decode(&case["source"]);
        let named=case["inputs"].as_array().unwrap().iter().map(|input|(input[0].as_str().unwrap().to_owned(),codec::decode(&input[1]))).collect::<Vec<_>>();
        let inputs=named.iter().map(|(name,instance)|BasicNamedSourceInput {name,instance}).collect::<Vec<_>>();
        let before=(codec::encode(&source),named.iter().map(|(name,tree)|json!([name,codec::encode(tree)])).collect::<Vec<_>>());
        for call in case["calls"].as_array().unwrap() {
            let selection=call["selection"].as_str().unwrap();let prefix=format!("{ordinal:02}-{selection}");
            let identity=proof::identity(&contract,ordinal,case,call,&prefix);
            let raw_before=raw_inputs(&source,&named);
            write(&directory,&format!("{prefix}-COMPLETE-INVOCATION.json"),&serde_json::to_vec_pretty(&identity)?)?;
            proof::write_call(&directory,&prefix,&identity,"COMPLETE-INPUT.json",&serde_json::to_vec_pretty(&json!({"before_raw":raw_before,"source_recipe":case["source"],"inputs_recipe":case["inputs"]}))?)?;
            proof::write_call(&directory,&prefix,&identity,"COMPLETE-INPUT.txt",format!("source={source:#?}\ninputs={named:#?}\nlimits={limits:#?}\ncontext: path=itemat473.mapping; main=itemat473.mapping; time=None; parameters=None; loader=None; cancellation=None\ncall={call:#?}\n").as_bytes())?;
            let observed=match case["model"].as_u64().unwrap() {
@DISPATCH@
                _=>panic!("authored model"),
            }?;
            let matched=observed.is_some_and(|record|record["matched"]==true);
            checks.push(json!({"invocation":identity,"matched":matched}));
        }
    }
    assert_eq!(checks.len(),@CORE_CALLS@);
    checks.extend(legacy_control::run(&directory,&contract)?);
    write(&directory,"ALL-COMPLETE-COMPARISONS.json",&serde_json::to_vec_pretty(&checks)?)?;
    assert!(checks.iter().all(|row|row["matched"]==true),"itemat473 differs; complete originals retained");
    Ok(())
}
