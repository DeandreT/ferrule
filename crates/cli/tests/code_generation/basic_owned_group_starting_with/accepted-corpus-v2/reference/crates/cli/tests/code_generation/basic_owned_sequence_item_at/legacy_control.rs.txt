use serde_json::json;
use std::path::Path;

pub fn run(directory: &Path) -> Result<bool, Box<dyn std::error::Error>> {
    let source = include_str!("legacy-input.json");
    let expected = include_str!("legacy-expected.json");
    super::write(
        directory,
        "LEGACY-COMPLETE-AUTHORED-CONTROLS.txt",
        format!("SOURCE={source}\nEXPECTED={expected}\n").as_bytes(),
    )?;
    // Retain each call before starting the next. A mapping or observer panic is
    // a failed call, while required evidence I/O remains fatal.
    let all = super::codec::retain_outcome(
        std::panic::catch_unwind(|| legacy0::execute_json_outputs(source)),
        |suffix, bytes| super::write(directory, &format!("LEGACY-All-{suffix}"), bytes),
        |output| json!({
            "variant": "All", "primary": output.primary,
            "extras": output.extras.iter().map(|item| json!({
                "name": item.name, "document": item.document,
            })).collect::<Vec<_>>()
        }),
        |actual| json!({"matched": matches!(actual,
            Ok(output) if output.primary == expected && output.extras.is_empty())}),
    )?;
    let selected = super::codec::retain_outcome(
        std::panic::catch_unwind(|| legacy0::execute_json_selected_target(
            source, codegen_runtime::TargetSelection::Primary,
        )),
        |suffix, bytes| super::write(directory, &format!("LEGACY-Primary-{suffix}"), bytes),
        |output| match output {
            legacy0::SelectedJsonTargetOutput::Primary(document) =>
                json!({"variant": "Primary", "document": document}),
            legacy0::SelectedJsonTargetOutput::Named(output) =>
                json!({"variant": "Named", "name": output.name, "document": output.document}),
        },
        |actual| json!({"matched": matches!(actual,
            Ok(legacy0::SelectedJsonTargetOutput::Primary(document)) if document == expected)}),
    )?;
    let all_matched = all.is_some_and(|record| record["matched"] == true);
    let selected_matched = selected.is_some_and(|record| record["matched"] == true);
    super::write(
        directory,
        "LEGACY-COMPLETE-COMPARISONS.json",
        &serde_json::to_vec_pretty(&json!([
            {"selection": "All", "matched": all_matched},
            {"selection": "Primary", "matched": selected_matched},
        ]))?,
    )?;
    Ok(all_matched && selected_matched)
}
