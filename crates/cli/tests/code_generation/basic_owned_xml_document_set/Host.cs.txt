// Generated C# host for the owned XML document-set cohort. It calls every
// case of one model through the text and byte entry points and records each
// outcome as a portable recipe; the test compares the recipes with the controls.
using System.Globalization;
using System.Text;
using System.Text.Json;
using System.Text.Json.Nodes;
using Ferrule.Generated;
using Ferrule.Runtime;

internal static class Driver
{
    private static JsonNode? Json(object? value) => JsonSerializer.SerializeToNode(value);

    private static JsonNode Encode(FerruleInstance instance)
    {
        if (instance is FerruleGroup group) return new JsonArray(Json("Group"), Json(group.Fields.Select(field => new JsonArray(Json(field.Name), Encode(field.Value))).ToArray()),
            Json(new { kind = group.XmlTypeOrigin.Kind.ToString(), identity = group.XmlTypeOrigin.Identity, literal = group.XmlTypeOrigin.Literal }));
        if (instance is FerruleRepeated repeated) return new JsonArray(Json("Repeated"), Json(repeated.Items.Select(Encode).ToArray()));
        if (instance is FerruleDocumentSet documents) return new JsonArray(Json("DocumentSet"), Json(documents.Documents.Select(document =>
            new JsonArray(Json(document.Path), Json(document.ResolvedSourcePath), Encode(document.Value))).ToArray()));
        if (instance is not FerruleScalar scalar) throw new InvalidOperationException("Closed authored tree kind.");
        var value = scalar.Value;
        return value.Kind switch {
            FerruleValueKind.Null => new JsonArray(Json("Null")), FerruleValueKind.JsonNull => new JsonArray(Json("JsonNull")), FerruleValueKind.XmlNil => new JsonArray(Json("XmlNil")),
            FerruleValueKind.String => new JsonArray(Json("String"), Json(value.StringValue)), FerruleValueKind.Bool => new JsonArray(Json("Bool"), Json(value.BooleanValue)),
            FerruleValueKind.Int64 => new JsonArray(Json("Int"), Json(value.Int64Value)),
            FerruleValueKind.Double => new JsonArray(Json("Float"), Json(unchecked((ulong)BitConverter.DoubleToInt64Bits(value.DoubleValue)).ToString("x16", CultureInfo.InvariantCulture))),
            _ => throw new InvalidOperationException("Closed typed output tag."),
        };
    }

    private static (string Path, string? Resolved, string Text)[] Members(JsonNode recipe) =>
        recipe.AsArray().Select(member => (member![0]!.GetValue<string>(), member[1]?.GetValue<string>(), member[2]!.GetValue<string>())).ToArray();

    private static JsonNode Runtime(FerruleRuntimeException runtime)
    {
        if (runtime.Error is FerruleRuntimeError.UnexpectedNamedSource or FerruleRuntimeError.DuplicateNamedSource)
            return new JsonObject { ["kind"] = runtime.Error.ToString(), ["name"] = runtime.Detail };
        if (runtime.InnerException is FerruleOwnedSourceException owned &&
            owned.Error is FerruleOwnedSourceErrorKind.SourceUnavailable or FerruleOwnedSourceErrorKind.MissingDocumentPath)
        {
            var recipe = new JsonObject { ["kind"] = "OwnedSource",
                ["owner"] = owned.Owner is null ? null : owned.Owner.IsPrimary ? "Primary" : owned.Owner.Name, ["cause"] = owned.Error.ToString() };
            if (runtime.Node is uint node) recipe["node"] = node;
            return recipe;
        }
        return new JsonObject { ["kind"] = "Other", ["debug"] = runtime.ToString() };
    }

    private static JsonNode Failure(Exception error)
    {
        if (error is not FerruleXmlDocumentSetInputException input)
            return new JsonObject { ["kind"] = "Unexpected", ["type"] = error.GetType().FullName, ["debug"] = error.ToString() };
        if (input.Member is { } member)
            return new JsonObject { ["kind"] = "Member", ["owner"] = member.Source.Kind == FerruleXmlInputSourceKind.Primary ? "Primary" : member.Source.Name,
                ["index"] = member.Index, ["path"] = member.Path, ["boundary"] = input.Boundary.Kind.ToString() };
        if (input.Boundary.Kind == FerruleXmlBoundaryErrorKind.Mapping && input.Boundary.InnerException is FerruleRuntimeException runtime)
            return new JsonObject { ["kind"] = "Mapping", ["error"] = Runtime(runtime) };
        return new JsonObject { ["kind"] = "Boundary", ["boundary"] = input.Boundary.Kind.ToString(), ["detail"] = input.Boundary.Message };
    }

    private static JsonNode Trees(IEnumerable<(string? Name, FerruleInstance Tree)> trees) =>
        new JsonObject { ["Ok"] = new JsonArray(trees.Select(tree => (JsonNode)new JsonArray(Json(tree.Name), Encode(tree.Tree))).ToArray()) };

    private static JsonNode Outputs(ExecutionOutputs output) =>
        Trees(new[] { ((string?)null, output.Primary) }.Concat(output.Extras.Select(extra => ((string?)extra.Name, extra.Instance))));

    private static JsonNode Selected(SelectedTargetOutput output) => output switch {
        SelectedTargetOutput.Primary primary => Trees(new[] { ((string?)null, primary.Instance) }),
        SelectedTargetOutput.Named named => Trees(new[] { ((string?)named.Output.Name, named.Output.Instance) }),
        _ => throw new InvalidOperationException("Closed selected output."),
    };

    private static JsonNode Invoke(string selection, (string Path, string? Resolved, string Text)[] source,
        (string Name, (string Path, string? Resolved, string Text)[] Members)[] inputs, bool bytes,
        FerruleOwnedSourceLimits limits, FerruleExecutionContext execution)
    {
        var target = selection == "Primary" ? FerruleTargetSelection.Primary() : FerruleTargetSelection.Named(selection);
        try
        {
            if (bytes)
            {
                FerruleXmlBytesDocumentSetMember[] Convert((string Path, string? Resolved, string Text)[] members) =>
                    members.Select(member => new FerruleXmlBytesDocumentSetMember(member.Path, member.Resolved, Encoding.UTF8.GetBytes(member.Text))).ToArray();
                var named = inputs.Select(input => new FerruleNamedXmlBytesDocumentSet(input.Name, Convert(input.Members))).ToArray();
                return selection == "All"
                    ? Outputs(GeneratedMapping.ExecuteXmlBytesDocumentSetsWithBasicOwnedSources(Convert(source), named, limits, execution))
                    : Selected(GeneratedMapping.ExecuteSelectedXmlBytesDocumentSetsWithBasicOwnedSources(Convert(source), target, named, limits, execution));
            }
            else
            {
                FerruleXmlDocumentSetMember[] Convert((string Path, string? Resolved, string Text)[] members) =>
                    members.Select(member => new FerruleXmlDocumentSetMember(member.Path, member.Resolved, member.Text)).ToArray();
                var named = inputs.Select(input => new FerruleNamedXmlDocumentSet(input.Name, Convert(input.Members))).ToArray();
                return selection == "All"
                    ? Outputs(GeneratedMapping.ExecuteXmlDocumentSetsWithBasicOwnedSources(Convert(source), named, limits, execution))
                    : Selected(GeneratedMapping.ExecuteSelectedXmlDocumentSetsWithBasicOwnedSources(Convert(source), target, named, limits, execution));
            }
        }
        catch (Exception error) { return new JsonObject { ["Err"] = Failure(error) }; }
    }

    private static int Main(string[] args)
    {
        if (args.Length != 3) throw new ArgumentException("controls, model and output directory");
        var controls = JsonNode.Parse(File.ReadAllText(args[0]))!;
        var model = int.Parse(args[1], CultureInfo.InvariantCulture);
        Directory.CreateDirectory(args[2]);
        var limits = new FerruleOwnedSourceLimits(16, 32, 64, 16_777_216, 512);
        var execution = new FerruleExecutionContext("xmldocset995.mapping");
        var rows = new JsonArray();
        foreach (var case_ in controls["cases"]!.AsArray())
        {
            if (case_!["model"]!.GetValue<int>() != model) continue;
            var source = Members(case_["source"]!);
            var inputs = case_["inputs"]!.AsArray().Select(input => (input![0]!.GetValue<string>(), Members(input[1]!))).ToArray();
            foreach (var call in case_["calls"]!.AsArray())
            {
                var selection = call!["selection"]!.GetValue<string>();
                foreach (var bytes in new[] { false, true })
                    rows.Add(new JsonObject { ["case"] = case_["id"]!.DeepClone(), ["selection"] = selection, ["bytes"] = bytes,
                        ["observed"] = Invoke(selection, source, inputs, bytes, limits, execution) });
            }
        }
        File.WriteAllText(Path.Combine(args[2], "OBSERVATIONS.json"), rows.ToJsonString(new JsonSerializerOptions { WriteIndented = true }));
        return 0;
    }
}
