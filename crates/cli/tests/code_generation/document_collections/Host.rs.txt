use codegen_runtime::{DocumentMember, ExecutionContext, Instance, RuntimeError, Value, XmlBoundaryErrorKind};
use ferrule_generated_mapping as generated;
use serde_json::{Value as Json, json};
use std::{error::Error, path::{Path, PathBuf}};

fn expected(case: &Json) -> Instance {
    Instance::DocumentSet(case["documents"].as_array().unwrap().iter().map(|d| {
        DocumentMember::new(d["path"].as_str().unwrap(), Instance::Group(vec![("Person".into(),
            Instance::Repeated(d["names"].as_array().unwrap().iter().map(|name| Instance::Group(vec![(
                "Name".into(), Instance::Scalar(Value::String(name.as_str().unwrap().into()))
            )].into())).collect()))].into())).unwrap()
    }).collect())
}

fn physical(xml: &str) -> Json {
    fn element(node: roxmltree::Node<'_, '_>) -> Json {
        assert!(node.children().all(|n| n.is_element() || n.is_text()));
        let children=node.children().filter(|n| n.is_element()).map(element).collect::<Vec<_>>();
        let text=node.children().filter(|n| n.is_text()).filter_map(|n| n.text()).collect::<String>();
        let text=if children.is_empty() {Some(text)} else {assert!(text.trim().is_empty());None};
        json!({"name":node.tag_name().name(),"namespace":node.tag_name().namespace(),
            "attributes":node.attributes().map(|a|(a.namespace(),a.name(),a.value())).collect::<Vec<_>>(),"text":text,"children":children})
    }
    element(roxmltree::Document::parse(xml).unwrap().root_element())
}

fn main() -> Result<(), Box<dyn Error>> {
    let root=PathBuf::from(std::env::args_os().nth(1).unwrap());
    let cases:Json=serde_json::from_slice(&std::fs::read(root.join("cases.json"))?)?;
    let schema=std::fs::read_to_string(root.join("source-schema.json"))?;
    let context=ExecutionContext::new(Path::new("collection-fixture"));
    for case in cases.as_array().unwrap() {
        let input=case["input"].as_str().unwrap();
        let source=codegen_runtime::parse_structured_xml_bytes(&schema,input.as_bytes())?;
        let direct=generated::execute(&source);
        std::fs::write(root.join(format!("rust-{}-typed.txt",case["id"].as_str().unwrap())),format!("{direct:#?}"))?;
        if case.get("error").is_some() {
            assert_eq!(direct,Err(RuntimeError::MappingException {node:12,message:Some("late document failure".into())}));
        } else {assert_eq!(direct?,expected(case));}
        for api in 0..4 {
            let result=match api {
                0=>generated::execute_xml_documents(input).map(|v|v.into_iter().map(|d|(d.path,d.document)).collect::<Vec<_>>()),
                1=>generated::execute_xml_documents_with_context(input,&context).map(|v|v.into_iter().map(|d|(d.path,d.document)).collect()),
                2=>generated::execute_xml_bytes_documents(input.as_bytes()).map(|v|v.into_iter().map(|d|(d.path,String::from_utf8(d.document).unwrap())).collect()),
                _=>generated::execute_xml_bytes_documents_with_context(input.as_bytes(),&context).map(|v|v.into_iter().map(|d|(d.path,String::from_utf8(d.document).unwrap())).collect()),
            };
            std::fs::write(root.join(format!("rust-{}-{api}.txt",case["id"].as_str().unwrap())),format!("{result:#?}"))?;
            if case.get("error").is_some() {
                let error=result.unwrap_err();
                assert!(error.member.is_none());
                assert_eq!(error.boundary.kind,XmlBoundaryErrorKind::Mapping);
                assert!(matches!(error.boundary.source().and_then(|e|e.downcast_ref::<RuntimeError>()),Some(RuntimeError::MappingException {node:12,..})));
            } else {
                let actual=result?;let wanted=case["documents"].as_array().unwrap();
                assert_eq!(actual.len(),wanted.len());
                for ((path,xml),document) in actual.iter().zip(wanted) {
                    assert_eq!(path,document["path"].as_str().unwrap());
                    let mut expected=String::from("<Output>");
                    for name in document["names"].as_array().unwrap() {expected.push_str(&format!("<Person><Name>{}</Name></Person>",name.as_str().unwrap()));}
                    expected.push_str("</Output>");assert_eq!(physical(xml),physical(&expected));
                }
            }
        }
    }
    Ok(())
}
