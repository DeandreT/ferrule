static class GeneratedMapping
{
    private static T Observe<T>(string api, Func<T> invoke)
    {
        var outcome = "error";
        var path = Environment.GetEnvironmentVariable("FERRULE_MAPPED_PUBLIC_CALLS");
        try { var value = invoke(); outcome = "ok"; if (path is not null) File.AppendAllText(path + ".jsonl", System.Text.Json.JsonSerializer.Serialize(new { api, outcome, value }) + "\n"); return value; }
        catch (Exception error) { if (path is not null) File.AppendAllText(path + ".jsonl", System.Text.Json.JsonSerializer.Serialize(new { api, outcome, type = error.GetType().FullName, detail = error.ToString() }) + "\n"); throw; }
        finally { if (path is not null) File.AppendAllText(path, $"{api}|{outcome}\n"); }
    }
    public static IReadOnlyList<FerruleCsvBytesDocument> ExecuteMappedCsvDocumentsBytes(FerruleInstance source) => Observe("bytes-default", () => ImplementationMapping.ExecuteMappedCsvDocumentsBytes(source));
    public static IReadOnlyList<FerruleCsvBytesDocument> ExecuteMappedCsvDocumentsBytes(FerruleInstance source, FerruleExecutionContext context) => Observe("bytes-context", () => ImplementationMapping.ExecuteMappedCsvDocumentsBytes(source, context));
    public static IReadOnlyList<FerruleCsvTextDocument> ExecuteMappedCsvDocuments(FerruleInstance source) => Observe("text-default", () => ImplementationMapping.ExecuteMappedCsvDocuments(source));
    public static IReadOnlyList<FerruleCsvTextDocument> ExecuteMappedCsvDocuments(FerruleInstance source, FerruleExecutionContext context) => Observe("text-context", () => ImplementationMapping.ExecuteMappedCsvDocuments(source, context));
    public static FerruleInstance Execute(FerruleInstance source, FerruleExecutionContext context) => ImplementationMapping.Execute(source, context);
}
