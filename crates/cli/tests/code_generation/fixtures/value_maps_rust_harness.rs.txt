use codegen_runtime::{Instance, Value, field, group, scalar};
use ferrule_generated_mapping::{execute, execute_with_sources};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let input = group(Vec::new());
    let expected = expected();
    std::fs::write("ordinary-host-input.txt", format!("{input:#?}\n"))?;
    std::fs::write("ordinary-host-literal-expected.txt", format!("{expected:#?}\n"))?;
    for (route, with_sources) in [("typed", false), ("typed-empty-named", true)] {
        let result = if with_sources { execute_with_sources(&input, &[]) } else { execute(&input) };
        std::fs::write(format!("ordinary-host-{route}-raw.txt"), format!("{result:#?}\n"))?;
        assert_eq!(result?, expected, "ordinary marker route {route}");
    }
    Ok(())
}

fn expected() -> Instance {
    group([
        field("Duplicate", scalar(Value::String("first".into()))),
        field("Default", scalar(Value::String("fallback".into()))),
        field("NoDefault", scalar(Value::Null)),
        field("FloatString", scalar(Value::String("float-string".into()))),
        field("Int", scalar(Value::String("int".into()))),
        field("Float", scalar(Value::String("float".into()))),
        field("Bool", scalar(Value::String("bool".into()))),
        field("Failed", scalar(Value::String("retained".into()))),
        field("Null", scalar(Value::String("null".into()))),
        field("XmlNil", scalar(Value::String("xml-nil".into()))),
        field("TypedJsonNull", scalar(Value::String("null-row".into()))),
        field("UntypedJsonNull", scalar(Value::String("json-null-row".into()))),
        field("TypedXmlNil", scalar(Value::String("xml-nil-row".into()))),
        field("UntypedXmlNil", scalar(Value::String("xml-nil-row".into()))),
        field("FunctionConstantJsonNull", scalar(Value::String("json-null-row".into()))),
        field("FunctionParameterJsonNull", scalar(Value::String("json-null-row".into()))),
        field("TypedNull", scalar(Value::String("null-first".into()))),
        field("UntypedNull", scalar(Value::String("null-first".into()))),
        field("FunctionParameterNull", scalar(Value::String("null-row".into()))),
        field("FunctionParameterXmlNil", scalar(Value::String("xml-nil-row".into()))),
    ])
}
