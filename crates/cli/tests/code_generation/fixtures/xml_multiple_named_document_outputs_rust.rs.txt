const CASES:[&str;10]=["ordered","empty-alpha","empty-beta","empty-both","selected","late-beta-output","primary-output","alpha-output","mapping-primary","mapping-alpha"];
fn named(value:&Instance)->Json {let Instance::DocumentSet(documents)=value else{panic!("complete named DocumentSet")};json!(documents.iter().map(|d|json!({"path":d.path(),"instance":snapshot::snapshot(d.value())})).collect::<Vec<_>>())}
fn physical(xml: &str) -> Json {
    fn element(node: roxmltree::Node<'_, '_>) -> Json {
        let mut attributes = node.attributes().map(|attribute|
            (attribute.namespace().unwrap_or(""), attribute.name(), attribute.value())).collect::<Vec<_>>();
        attributes.sort_by_key(|(namespace, name, _)| (*namespace, *name));
        let children = node.children().filter(|child| child.is_element()).map(element).collect::<Vec<_>>();
        let raw_text = node.children().filter(|child| child.is_text()).filter_map(|child| child.text()).collect::<String>();
        let text = if children.is_empty() { Some(raw_text) } else {
            assert!(raw_text.chars().all(|character| matches!(character, ' ' | '\t' | '\r' | '\n')), "only element-only formatting whitespace");
            None
        };
        json!({"name":node.tag_name().name(),"namespace":node.tag_name().namespace().unwrap_or(""),"text":text,
            "attributes":attributes.iter().map(|(namespace,name,value)| json!({"name":name,"namespace":namespace,"value":value})).collect::<Vec<_>>(),"children":children})
    }
    element(roxmltree::Document::parse(xml).expect("complete well-formed output").root_element())
}
fn boundary(error: &XmlBoundaryError) -> Json {
    let mut causes = Vec::new();
    let mut current = error.source();
    while let Some(cause) = current {
        causes.push(json!({"detail":cause.to_string(),"debug":format!("{cause:?}")}));
        current = cause.source();
    }
    json!({"kind":format!("{:?}",error.kind),"detail":error.detail,"bytes":error.bytes,"limit":error.limit,"causes":causes})
}
type Mixed = (String, Vec<(String, Vec<(String, String)>)>);
fn call(api: usize, text: &str, bytes: &[u8], execution: &ExecutionContext<'_>) -> Result<Mixed, XmlDocumentOutputsExecutionError> {
    fn strings(output: generated::XmlDocumentExecutionOutputs) -> Mixed {
        (output.primary, output.extras.into_iter().map(|extra| (extra.name.to_owned(), extra.documents.into_iter().map(|document| (document.path, document.document)).collect())).collect())
    }
    fn buffers(output: generated::XmlBytesDocumentExecutionOutputs) -> Mixed {
        (String::from_utf8(output.primary).expect("owned primary UTF-8"), output.extras.into_iter().map(|extra| (extra.name.to_owned(), extra.documents.into_iter().map(|document| (document.path, String::from_utf8(document.document).expect("owned member UTF-8"))).collect())).collect())
    }
    match api {
        0 => generated::execute_xml_document_outputs(text).map(strings),
        1 => generated::execute_xml_document_outputs_with_context(text, execution).map(strings),
        2 => generated::execute_xml_bytes_document_outputs(bytes).map(buffers),
        3 => generated::execute_xml_bytes_document_outputs_with_context(bytes, execution).map(buffers),
        _ => panic!("one of four mixed APIs"),
    }
}

        let mut original_output=None;let mut original_mapping=None;

                        codegen_runtime::serialize_xml_document(&schemas[index],documents[1].value(),true,true,None,Some(&hints[index])).unwrap_err()
                    });
                }
            }
            Err(error)=>{assert!(case.starts_with("mapping-"));assert!(matches!(error,RuntimeError::EmptyDynamicTargetPath{node:10}));original_mapping=Some((error.to_string(),format!("{error:?}")));println!("{}",json!({"case":case,"api":-1,"source":snapshot::snapshot(&parsed),"primary":null,"extras":null,"mapping_failure":true}));}
        }
        for api in 0..4 {
            match call(api,text,&bytes,&execution) {
                Ok((primary,extras))=>{
                    assert!(matches!(case,"ordered"|"empty-alpha"|"empty-beta"|"empty-both"|"selected"));assert_eq!(extras.len(),2);for ((name,_),expected) in extras.iter().zip(&names){assert_eq!(name,expected);}
                    let primary_physical=physical(&primary);assert_eq!(primary_physical,physical(&std::fs::read_to_string(directory.join(format!("native-{case}-primary.xml")))?));
                    let extras=extras.into_iter().enumerate().map(|(declaration_index,(name,documents))|json!({"name":name,"documents":documents.into_iter().enumerate().map(|(index,(path,document))|{let actual=physical(&document);let native=physical(&std::fs::read_to_string(directory.join(format!("native-{case}-{declaration_index}-{index}.xml"))).unwrap());assert_eq!(actual,native);json!({"path":path,"document":document,"physical":actual,"native_physical_match":true})}).collect::<Vec<_>>()})).collect::<Vec<_>>();
                    println!("{}",json!({"case":case,"api":api,"kind":"ok","primary":{"document":primary,"physical":primary_physical,"native_physical_match":true},"extras":extras}));
                }
                Err(error)=>{
                    assert!(std::ptr::eq(error.source().unwrap().downcast_ref::<XmlBoundaryError>().unwrap(),error.boundary.as_ref()));assert_eq!((error.boundary.bytes,error.boundary.limit),(None,None));
                    let (owner,runtime_error,node)=if matches!(case,"late-beta-output"|"alpha-output"|"primary-output") {

                            let Some(XmlDocumentOutputsOwner::NamedMember{declaration_index,name,index,path:actual_path})=&error.owner else{panic!("named member owner")};assert_eq!((*declaration_index,name.as_str(),*index,actual_path.as_str()),(declaration,wanted,1,path));json!({"kind":"NamedMember","declaration_index":declaration_index,"name":name,"index":index,"path":actual_path})

                    println!("{}",json!({"case":case,"api":api,"kind":format!("{:?}",error.boundary.kind),"wrapper":"XmlDocumentOutputsExecutionError","owner":owner,"boundary":boundary(&error.boundary),"original_boundary":original_output.as_ref().map(boundary),"boundary_identity":true,"original_boundary_match":true,"runtime_error":runtime_error,"node":node,"no_partial_result":true}));
                }
            }
        }
    }Ok(())
}
