use codegen_runtime::JsonBoundaryError;
use ferrule_generated_mapping::{execute_json, execute_json_bytes};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let cases: Vec<serde_json::Value> = serde_json::from_str(include_str!("../cases.json"))?;
    for case in cases {
        if let Some(input) = case.get("input").and_then(serde_json::Value::as_str) {
            verify(execute_json(input), &case);
            verify(
                execute_json_bytes(input.as_bytes()).map(|bytes| String::from_utf8(bytes).unwrap()),
                &case,
            );
        } else {
            let bytes: Vec<u8> = serde_json::from_value(case["bytes"].clone())?;
            verify(
                execute_json_bytes(&bytes).map(|bytes| String::from_utf8(bytes).unwrap()),
                &case,
            );
        }
    }
    Ok(())
}

fn verify(result: Result<String, JsonBoundaryError>, case: &serde_json::Value) {
    if case["reject"] == true {
        assert!(matches!(result, Err(JsonBoundaryError::InvalidInput { .. })), "{case}: {result:?}");
    } else {
        let actual: serde_json::Value = serde_json::from_str(&result.unwrap()).unwrap();
        let expected = match case.get("expected_json").and_then(serde_json::Value::as_str) {
            Some(text) => serde_json::from_str(text).unwrap(),
            None => case["expected"].clone(),
        };
        assert_eq!(actual, expected, "{case}");
    }
}
