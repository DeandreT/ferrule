    private static object? NormalizeProperty(object? value)
    {
        if (value is null) return null;
        if (value is Exception e) return new { type = e.GetType().FullName, message = e.Message, reference_id = ReferenceId(e) };
        if (value is byte[] bytes) return Buffer(bytes);
        if (value is char codeUnit) return new { utf16_code_unit = (int)codeUnit }; // Retain lone surrogate exception fields losslessly.
        if (value is Enum enumeration) return enumeration.ToString();
        if (value is UInt128 large) return large.ToString(CultureInfo.InvariantCulture);
        if (value is Int128 signed) return signed.ToString(CultureInfo.InvariantCulture);
        if (value is string || value.GetType().IsPrimitive || value is decimal) return value;
        if (value is MemberInfo member) return member.ToString();
        if (value is IEnumerable enumerable) { var list = new List<object?>(); foreach (var item in enumerable) list.Add(NormalizeProperty(item)); return list; }
        var properties = new SortedDictionary<string, object?>(StringComparer.Ordinal);
        foreach (var p in value.GetType().GetProperties(BindingFlags.Public | BindingFlags.Instance).Where(p => p.CanRead && p.GetIndexParameters().Length == 0))
        {
            try { properties[p.Name] = NormalizeProperty(p.GetValue(value)); }
            catch (Exception getterError) { properties[p.Name] = new { getter_error_type = getterError.GetType().FullName, getter_error_message = getterError.Message }; }
        }
        return properties;
    }
    private static object ExceptionProperties(Exception error, bool semanticOnly)
    {

        foreach (var p in error.GetType().GetProperties(BindingFlags.Public | BindingFlags.Instance).Where(p => p.CanRead && p.GetIndexParameters().Length == 0 && p.Name != "InnerException"))

            catch (Exception e) { fields[p.Name] = new { getter_error_type = e.GetType().FullName, getter_error_message = e.Message }; }

        return fields;
    }

        XmlDocumentExecutionOutputs? textResult = null; XmlBytesDocumentExecutionOutputs? byteResult = null;
        try

            row["outputs"] = byteApi ? OriginalByteOutputs(byteResult!) : OriginalTextOutputs(textResult!);
        // This dedicated original channel has no optional parser, physical-tree or DTO-reflection observations.
        var originalRecord = Emit(row);

                statics_transport = supplied.Select(input => new { name = input.Name,
                    original_declaration_index = _project.GetProperty("extra_sources").EnumerateArray().Select((source, index) => (source, index)).Single(pair => pair.source.GetProperty("name").GetString() == input.Name).index,
                    buffer = Buffer(input.Bytes), diagnostic_parse_outside_product = TypedObservation(NamedSchema(input.Name), input.Bytes) }).ToArray() };
            observed["callbacks"] = loader.Callbacks.Select(call =>

            }).ToArray();
            if (Equals(row["outcome"], "returned_complete_outputs"))
            {

                try { observed["original_returned_value"] = NormalizeProperty(byteApi ? (object?)byteResult : textResult); }
                catch (Exception error) { observed["original_value_recording_error"] = ExceptionSnapshot(error); }
                observed["returned_ownership"] = ReturnOwnership(textResult, byteResult);

                catch (Exception error) { observed["diagnostic_recording_error"] = ExceptionSnapshot(error); }
            }
            if (loader.Thrown is not null) observed["original_host_exception"] = ExceptionSnapshot(loader.Thrown);
            if (row["error"] is not null && JsonSerializer.SerializeToElement(row["error"], JsonOptions).TryGetProperty("owner", out var owner) && owner.ValueKind == JsonValueKind.Object && owner.TryGetProperty("Output", out _))

            return observed;
        });
    }

    private static object ReturnOwnership(XmlDocumentExecutionOutputs? text, XmlBytesDocumentExecutionOutputs? bytes)
    {
        if (bytes is not null)
        {
            var buffers = new HashSet<byte[]>(ReferenceEqualityComparer.Instance); var distinct = buffers.Add(bytes.Primary);
            foreach (var extra in bytes.Extras) foreach (var member in extra.Documents) distinct &= buffers.Add(member.Document);
            return new { extras_readonly = bytes.Extras is IList byteExtrasList && byteExtrasList.IsReadOnly,
                documents_readonly = bytes.Extras.Select(extra => extra.Documents is IList members && members.IsReadOnly).ToArray(),
                all_buffers_distinct = distinct, actual_buffer_count = buffers.Count,
                expected_buffer_count = 1 + bytes.Extras.Sum(extra => extra.Documents.Count) };
        }
        if (text is not null) return new { extras_readonly = text.Extras is IList textExtrasList && textExtrasList.IsReadOnly,
            documents_readonly = text.Extras.Select(extra => extra.Documents is IList members && members.IsReadOnly).ToArray() };
        return new { not_returned = true };
    }
    private static FerruleInstance RestoreOrdinary(JsonElement snapshot)
    {
        switch (snapshot.GetProperty("kind").GetString())
        {
            case "Scalar":
                var tag = snapshot.GetProperty("tag").GetString();
                return new FerruleScalar(tag switch {
                    "Null" => FerruleValue.Null, "JsonNull" => FerruleValue.JsonNull, "XmlNil" => FerruleValue.XmlNil,
                    "Bool" => FerruleValue.FromBoolean(snapshot.GetProperty("value").GetBoolean()),
                    "Int" => FerruleValue.FromInt64(snapshot.GetProperty("value").GetInt64()),
                    "Float" => FerruleValue.FromDouble(BitConverter.Int64BitsToDouble(unchecked((long)ulong.Parse(snapshot.GetProperty("bits_hex").GetString()!, NumberStyles.HexNumber, CultureInfo.InvariantCulture)))),
                    "String" => FerruleValue.FromString(snapshot.GetProperty("value").GetString()!),
                    _ => throw new InvalidDataException("Unknown authored scalar snapshot " + tag) });
            case "Group":
                if (snapshot.GetProperty("xml_type_origin").GetProperty("kind").GetString() != "Unknown") throw new InvalidDataException("Ordinary fixture restoration refuses non-Unknown XML origin");
                return new FerruleGroup(snapshot.GetProperty("fields").EnumerateArray().Select(field => new FerruleField(field.GetProperty("name").GetString()!, RestoreOrdinary(field.GetProperty("instance")))));
            case "Repeated": return new FerruleRepeated(snapshot.GetProperty("items").EnumerateArray().Select(RestoreOrdinary));
            default: throw new InvalidDataException("Unknown ordinary member snapshot shape");
        }
    }

        var native = _nativeRows.SingleOrDefault(row => row.GetProperty("case").GetString() == c.GetProperty("id").GetString());
        if (native.ValueKind == JsonValueKind.Undefined) return new { outcome = "missing_original_native" };
        var output = owner.GetProperty("Output"); JsonElement schema, options, snapshot;
        if (output.ValueKind == JsonValueKind.String && output.GetString() == "Primary")
        { schema = _project.GetProperty("target"); options = _project.GetProperty("target_options"); snapshot = native.GetProperty("complete_typed_outputs").GetProperty("primary"); }
        else
        { var member = output.GetProperty("NamedMember"); var declaration = member.GetProperty("declaration_index").GetInt32(); var index = member.GetProperty("index").GetInt32();
            var target = _project.GetProperty("extra_targets")[declaration]; schema = target.GetProperty("schema"); options = target.GetProperty("options");
            snapshot = native.GetProperty("complete_typed_outputs").GetProperty("extras")[declaration].GetProperty("instance").GetProperty("documents")[index].GetProperty("instance"); }
        var value = RestoreOrdinary(snapshot); var hints = options.TryGetProperty("xml_schema_hints", out var hint) ? hint.GetRawText() : null;
        try { var xml = FerruleXml.SerializeDocumentEmbedded(schema.GetRawText(), value, true, true, null, hints);
            return new { outcome = "returned", buffer = Buffer(StrictUtf8.GetBytes(xml)), original_snapshot = snapshot, schema, options }; }
        catch (FerruleXmlBoundaryException boundary) { return new { outcome = "error", boundary = Boundary(boundary), original_snapshot = snapshot, schema, options }; }
    }

    public static int Main(string[] args)
    {
        try
        {
            if (args.Length != 4) throw new ArgumentException("Expected fixtureRoot, rawProjectPath, caseManifestPath, nativeRecordsPath.");
            _root = Path.GetFullPath(args[0]); _variant = Path.GetFileNameWithoutExtension(args[1]); _projectBytes = File.ReadAllBytes(args[1]);
            using var project = JsonDocument.Parse(_projectBytes); _project = project.RootElement.Clone();
            var nativeBytes = File.ReadAllBytes(args[3]);
            _nativeOriginalStreamReference = new { path = args[3], size = nativeBytes.LongLength, sha256 = Convert.ToHexString(SHA256.HashData(nativeBytes)).ToLowerInvariant() };
            _nativeRows = StrictUtf8.GetString(nativeBytes).Split('\n', StringSplitOptions.RemoveEmptyEntries).Select(line => { using var document = JsonDocument.Parse(line); return document.RootElement.Clone(); }).ToArray();
            using var cases = JsonDocument.Parse(File.ReadAllBytes(args[2]));
            foreach (var c in cases.RootElement.GetProperty("cases").EnumerateArray())
            {
                if (!string.Equals(Path.GetFileName(c.GetProperty("project").GetString()), Path.GetFileName(args[1]), StringComparison.Ordinal) || !c.GetProperty("backends").EnumerateArray().Any(backend => backend.GetString() == "csharp")) continue;
                foreach (var method in c.GetProperty("methods").EnumerateArray()) RecordCall(c, method.GetString()!);
            }
            return 0;
        }
        catch (Exception error)
        {
            Console.Error.WriteLine(JsonSerializer.Serialize(new { schema = 1, backend = "csharp", variant = _variant, phase = "ordinary-caller-setup-or-recording", error = ExceptionSnapshot(error) }, JsonOptions)); Console.Error.Flush(); return 2;
        }
    }
