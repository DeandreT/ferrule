use codegen_runtime::JsonBoundaryError;
use ferrule_generated_mapping::{execute_json, execute_json_bytes};

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let cases: Vec<serde_json::Value> = serde_json::from_str(include_str!("../cases.json"))?;
    for case in &cases {
        if let Some(input) = case.get("input").and_then(serde_json::Value::as_str) {
            let saved_input = input.to_owned();
            verify(execute_json(input), case);
            let bytes = input.as_bytes().to_vec();
            let saved_bytes = bytes.clone();
            verify_bytes(execute_json_bytes(&bytes), case);
            assert_eq!(input, saved_input, "caller text changed: {case}");
            assert_eq!(bytes, saved_bytes, "caller UTF-8 changed: {case}");
        } else {
            let bytes: Vec<u8> = serde_json::from_value(case["bytes"].clone())?;
            let saved_bytes = bytes.clone();
            verify_bytes(execute_json_bytes(&bytes), case);
            assert_eq!(bytes, saved_bytes, "caller invalid UTF-8 changed: {case}");
        }
    }
    println!("verified {} ordinary JSON implicit occurrence cases", cases.len());
    Ok(())
}

fn verify_bytes(result: Result<Vec<u8>, JsonBoundaryError>, case: &serde_json::Value) {
    if case["reject"] == true {
        verify(result.map(|bytes| String::from_utf8(bytes).expect("output is UTF-8")), case);
        return;
    }
    let bytes = result.expect("successful generated byte output");
    assert_eq!(bytes, case["expected_json"].as_str().unwrap().as_bytes(),
        "literal ordered UTF-8 bytes differ: {case}");
    verify(Ok(String::from_utf8(bytes).expect("output is UTF-8")), case);
}

fn verify(result: Result<String, JsonBoundaryError>, case: &serde_json::Value) {
    if case["reject"] == true {
        let message = match result {
            Err(JsonBoundaryError::InvalidInput { message }) => message,
            other => panic!("expected input boundary error: {case}: {other:?}"),
        };
        assert!(!message.is_empty(), "input boundary detail missing: {case}");
        if case["invalid_utf8"] == true {
            assert!(message.contains("UTF-8"), "UTF-8 error differs: {case}: {message}");
        }
        return;
    }
    let output = result.expect("successful generated text output");
    assert_eq!(output, case["expected_json"].as_str().unwrap(),
        "literal ordered JSON text differs: {case}");
    let actual: serde_json::Value = serde_json::from_str(&output).unwrap();
    assert_eq!(actual, case["expected"], "complete output value differs: {case}");
    let batches = actual["Batches"].as_array().unwrap();
    assert_eq!(batches.len(), case["parent_count"].as_u64().unwrap() as usize, "{case}");
    for batch in batches {
        let records = batch["Records"].as_array().unwrap();
        assert_eq!(records.len(), 2, "{case}");
        assert_eq!(records[0]["Branch"], "left", "{case}");
        assert_eq!(records[1]["Branch"], "right", "{case}");
    }
}
