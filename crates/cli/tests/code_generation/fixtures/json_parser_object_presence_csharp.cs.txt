using System.Text;
using System.Text.Json;
using System.Text.Json.Nodes;
using Ferrule.Generated;
using Ferrule.Runtime;

using var cases = JsonDocument.Parse(File.ReadAllText("cases.json"));
foreach (var test in cases.RootElement.EnumerateArray())
{
    var parameters = new FerruleRuntimeParameters(test.GetProperty("parameters").EnumerateObject().Select(parameter =>
        KeyValuePair.Create(parameter.Name, parameter.Value.ValueKind switch
        {
            JsonValueKind.String => FerruleValue.FromString(parameter.Value.GetString()!),
            JsonValueKind.True => FerruleValue.FromBoolean(true),
            JsonValueKind.False => FerruleValue.FromBoolean(false),
            JsonValueKind.Number => FerruleValue.FromInt64(parameter.Value.GetInt64()),
            JsonValueKind.Null => FerruleValue.Null,
            JsonValueKind.Object when parameter.Value.GetProperty("non_finite").GetString() == "positive-infinity" => FerruleValue.FromDouble(double.PositiveInfinity),
            _ => throw new Exception($"Unsupported authored parameter: {test}"),
        })));
    var context = new FerruleExecutionContext("parser-input.ferrule.json", "parser-input.ferrule.json", null, parameters);
    var input = test.GetProperty("input").GetString()!;
    Verify(() => GeneratedMapping.ExecuteJson(input, context), test);
    Verify(() => Encoding.UTF8.GetString(GeneratedMapping.ExecuteJsonBytes(Encoding.UTF8.GetBytes(input), context)), test);
}
if (File.ReadAllText("design.txt") == "root")
{
    VerifyFunctionCases();
    VerifyDocumentSets();
}
Console.WriteLine($"verified {cases.RootElement.GetArrayLength()} parser object presence cases");

static void Verify(Func<string> run, JsonElement test)
{
    if (test.TryGetProperty("missing_parameter", out var missing))
    {
        try { _ = run(); }
        catch (FerruleRuntimeException error) when (error.Error == FerruleRuntimeError.MissingRuntimeParameter &&
            error.Node == missing.GetProperty("node").GetUInt32() && error.RuntimeParameter == missing.GetProperty("name").GetString())
        { return; }
        throw new Exception($"Missing parameter error differs: {test}");
    }
    if (test.TryGetProperty("mapping_exception", out var cardinality))
    {
        try { _ = run(); }
        catch (FerruleRuntimeException error) when (error.Error == FerruleRuntimeError.MappingException &&
            error.Node == cardinality.GetProperty("node").GetUInt32() && error.MappingExceptionMessage == cardinality.GetProperty("message").GetString())
        { return; }
        throw new Exception($"Cardinality refusal differs: {test}");
    }
    if (test.TryGetProperty("parameter_type_error", out var type))
    {
        try { _ = run(); }
        catch (FerruleRuntimeException error) when (error.Error == FerruleRuntimeError.RuntimeParameterType &&
            error.Node == type.GetProperty("node").GetUInt32() && error.RuntimeParameter == type.GetProperty("name").GetString() &&
            type.GetProperty("expected").GetString() == "string" && error.ExpectedScalarType == FerruleScalarType.String &&
            type.GetProperty("found").GetString() == "float" && error.FoundKind == FerruleValueKind.Double)
        { return; }
        throw new Exception($"Parameter type error differs: {test}");
    }
    if (test.TryGetProperty("execution_error", out var expectedError))
    {
        try { _ = run(); }
        catch (FerruleRuntimeException error) when (error.Error == FerruleRuntimeError.FunctionInvalidArgument &&
            error.Function == expectedError.GetProperty("function").GetString() && error.Detail == expectedError.GetProperty("message").GetString())
        { return; }
        throw new Exception($"JSON parser error differs: {test}");
    }
    if (test.TryGetProperty("reject", out _))
    {
        try { _ = run(); }
        catch (FerruleRuntimeException error) when (error.Error == FerruleRuntimeError.JsonBoundary) { return; }
        throw new Exception($"Invalid primary input was accepted: {test}");
    }
    var actual = run();
    if (!JsonNode.DeepEquals(JsonNode.Parse(actual), JsonNode.Parse(test.GetProperty("expected").GetRawText())))
    {
        throw new Exception($"JSON output differs: {test}; actual: {actual}");
    }
}

static void VerifyFunctionCases()
{
    using var cases = JsonDocument.Parse(File.ReadAllText("function-cases.json"));
    foreach (var test in cases.RootElement.EnumerateArray())
    {
        var args = test.GetProperty("arguments").EnumerateArray().Select(value => value.ValueKind switch
        {
            JsonValueKind.String => FerruleValue.FromString(value.GetString()!),
            JsonValueKind.Null => FerruleValue.Null,
            _ => throw new Exception("Unsupported authored argument"),
        }).ToArray();
        if (test.TryGetProperty("execution_error", out var expected))
        {
            try { _ = FerruleFunctions.Call("json_parse_object_exists", args); }
            catch (FerruleRuntimeException error) when (error.Error == FerruleRuntimeError.FunctionInvalidArgument &&
                error.Function == expected.GetProperty("function").GetString() && error.Detail == expected.GetProperty("message").GetString())
            { continue; }
            throw new Exception($"Function refusal differs: {test}");
        }
        var actual = FerruleFunctions.Call("json_parse_object_exists", args);
        if (actual.Kind != FerruleValueKind.Bool || actual.BooleanValue != test.GetProperty("expected").GetBoolean())
            throw new Exception($"Object presence differs: {test}");
    }
}

static void VerifyDocumentSets()
{
    for (var size = 0; size <= 2; size++)
    {
        foreach (var supplied in new[] { false, true })
        {
            var source = new FerruleDocumentSet(Enumerable.Range(0, size).Select(index =>
                new FerruleDocument($"authored-{index}.json", new FerruleGroup(Array.Empty<FerruleField>()))));
            var parameters = new FerruleRuntimeParameters(supplied
                ? new[] { KeyValuePair.Create("payload", FerruleValue.FromString("{\"Header\":{\"Replacement\":\"R\"}}")) }
                : Array.Empty<KeyValuePair<string, FerruleValue>>());
            var context = new FerruleExecutionContext("object-presence.ferrule.json", "object-presence.ferrule.json", null, parameters);
            if (size == 2)
            {
                try { _ = GeneratedMapping.Execute(source, context); }
                catch (FerruleRuntimeException error) when (error.Error == FerruleRuntimeError.MappingException && error.Node == 14 &&
                    error.MappingExceptionMessage == "JSON parser root object presence does not support multiple primary rows")
                { continue; }
                throw new Exception("Multiple typed documents did not refuse before bindings");
            }
            if (size == 1 && !supplied)
            {
                try { _ = GeneratedMapping.Execute(source, context); }
                catch (FerruleRuntimeException error) when (error.Error == FerruleRuntimeError.MissingRuntimeParameter && error.Node == 2 && error.RuntimeParameter == "payload")
                { continue; }
                throw new Exception("Singleton typed document did not require the parameter");
            }
            var output = GeneratedMapping.Execute(source, context) as FerruleGroup ?? throw new Exception("Object output expected");
            if (size == 0 && output.Fields.Count != 0) throw new Exception("Empty source did not keep empty First output");
            if (size == 1 && (!output.TryGetField("Label", out var label) || label is not FerruleScalar scalar ||
                scalar.Value.Kind != FerruleValueKind.String || scalar.Value.StringValue != "R")) throw new Exception("Singleton typed document output differs");
        }
    }
}
