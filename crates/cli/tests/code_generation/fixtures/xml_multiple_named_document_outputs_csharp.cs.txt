static Mixed Call(int api, string text, byte[] bytes, FerruleExecutionContext execution)
{
    static Mixed Strings(XmlDocumentExecutionOutputs output)
    {
        ReadOnly(output.Extras);
        foreach (var extra in output.Extras) ReadOnly(extra.Documents);
        return new(output.Primary, output.Extras.Select(extra => (extra.Name, extra.Documents.Select(document => (document.Path, document.Document)).ToArray())).ToArray());
    }
    static Mixed Buffers(XmlBytesDocumentExecutionOutputs output)
    {
        ReadOnly(output.Extras);
        var buffers = new HashSet<byte[]>(ReferenceEqualityComparer.Instance) { output.Primary };
        foreach (var extra in output.Extras)
        {
            ReadOnly(extra.Documents);
            foreach (var document in extra.Documents) Require(buffers.Add(document.Document), "all owned primary/member buffers distinct");
        }
        var decoder = new UTF8Encoding(false, true);
        return new(decoder.GetString(output.Primary), output.Extras.Select(extra => (extra.Name, extra.Documents.Select(document => (document.Path, decoder.GetString(document.Document))).ToArray())).ToArray());
    }
    return api switch
    {
        0 => Strings(GeneratedMapping.ExecuteXmlDocumentOutputs(text)),
        1 => Strings(GeneratedMapping.ExecuteXmlDocumentOutputs(text, execution)),
        2 => Buffers(GeneratedMapping.ExecuteXmlBytesDocumentOutputs(bytes)),
        3 => Buffers(GeneratedMapping.ExecuteXmlBytesDocumentOutputs(bytes, execution)),
        _ => throw new InvalidOperationException("one of four mixed APIs"),
    };
}
static void ReadOnly<T>(IReadOnlyList<T> values)
{
    if (values is IList<T> mutable)
    {
        try { mutable.Add(default!); throw new InvalidOperationException("returned collection permitted mutation"); }
        catch (NotSupportedException) { }
    }
}
static object Boundary(FerruleXmlBoundaryException error)
{
    var causes = new List<object>();
    for (var cause = error.InnerException; cause is not null; cause = cause.InnerException)
        causes.Add(new { type = cause.GetType().Name, detail = cause.Message });
    return new { kind = error.Kind.ToString(), detail = error.Detail, bytes = error.Bytes, limit = error.Limit, causes };
}
static object Physical(string xml)
{
    static object Element(XElement element)
    {
        var attributes = element.Attributes().Where(attribute => !attribute.IsNamespaceDeclaration)
            .OrderBy(attribute => attribute.Name.NamespaceName, StringComparer.Ordinal).ThenBy(attribute => attribute.Name.LocalName, StringComparer.Ordinal)
            .Select(attribute => new { name = attribute.Name.LocalName, @namespace = attribute.Name.NamespaceName, value = attribute.Value }).ToArray();
        var children = element.Elements().Select(Element).ToArray();
        var rawText = string.Concat(element.Nodes().OfType<XText>().Select(text => text.Value));
        string? text = rawText;
        if (children.Length != 0)
        {
            Require(rawText.All(character => character is ' ' or '\t' or '\r' or '\n'), "only element-only formatting whitespace");
            text = null;
        }
        return new { name = element.Name.LocalName, @namespace = element.Name.NamespaceName, text, attributes, children };
    }
    return Element(XDocument.Parse(xml, LoadOptions.PreserveWhitespace).Root ?? throw new InvalidOperationException("document root"));
}
static object Snapshot(FerruleInstance input) => input switch
{
    FerruleScalar scalar => Scalar(scalar.Value),
    FerruleGroup group => new { kind = "Group", xml_type_origin = Origin(group.XmlTypeOrigin), fields = group.Fields.Select(field => new { name = field.Name, instance = Snapshot(field.Value) }).ToArray() },
    FerruleRepeated repeated => new { kind = "Repeated", items = repeated.Items.Select(Snapshot).ToArray() },
    FerruleMappedSequence mapped => new { kind = "MappedSequence", items = mapped.Items.Select(Snapshot).ToArray() },
    _ => throw new InvalidOperationException("unexpected closed instance variant"),
};
static object Origin(FerruleXmlTypeOrigin origin) => origin.Kind switch
{
    FerruleXmlTypeOriginKind.Unknown => new { kind = "Unknown" },
    FerruleXmlTypeOriginKind.Absent => new { kind = "Absent" },
    FerruleXmlTypeOriginKind.Explicit => new { kind = "Explicit", identity = origin.Identity },
    FerruleXmlTypeOriginKind.ExplicitPadded => new { kind = "ExplicitPadded", literal = origin.Literal, resolved_identity = origin.Identity },
    _ => throw new InvalidOperationException("unexpected XML type origin"),
};
static object Scalar(FerruleValue value) => value.Kind switch
{
    FerruleValueKind.Null => new { kind = "Scalar", tag = "Null" },
    FerruleValueKind.JsonNull => new { kind = "Scalar", tag = "JsonNull" },
    FerruleValueKind.XmlNil => new { kind = "Scalar", tag = "XmlNil" },
    FerruleValueKind.Bool => new { kind = "Scalar", tag = "Bool", value = value.BooleanValue },
    FerruleValueKind.Int64 => new { kind = "Scalar", tag = "Int", value = value.Int64Value },
    FerruleValueKind.Double => new { kind = "Scalar", tag = "Float", bits_hex = BitConverter.DoubleToUInt64Bits(value.DoubleValue).ToString("x16", CultureInfo.InvariantCulture) },
    FerruleValueKind.String => new { kind = "Scalar", tag = "String", value = value.StringValue },
    _ => throw new InvalidOperationException("unexpected scalar tag"),
};
static void Require(bool condition, string detail)
{
    if (!condition) throw new InvalidOperationException(detail);
}
