 static bool JsonEqual(JsonElement value,JsonElement expected)
 {
  if(value.ValueKind!=expected.ValueKind)return false;
  switch(value.ValueKind)
  {
   case JsonValueKind.Object:
    var fields=value.EnumerateObject().ToDictionary(p=>p.Name,p=>p.Value,StringComparer.Ordinal);var other=expected.EnumerateObject().ToArray();
    return fields.Count==other.Length&&other.All(p=>fields.TryGetValue(p.Name,out var field)&&JsonEqual(field,p.Value));
   case JsonValueKind.Array:
    var values=value.EnumerateArray().ToArray();var others=expected.EnumerateArray().ToArray();return values.Length==others.Length&&values.Zip(others).All(p=>JsonEqual(p.First,p.Second));
   case JsonValueKind.String:return value.GetString()==expected.GetString();
   case JsonValueKind.Number:return value.GetRawText()==expected.GetRawText();
   case JsonValueKind.True:case JsonValueKind.False:case JsonValueKind.Null:return true;
   default:throw new InvalidOperationException("closed JSON owner value");
  }
 }
 static void ReadOnly<T>(IReadOnlyList<T> value)=>Require(value is IList<T> mutable&&mutable.IsReadOnly,"actual readonly returned list");

 static (string Path,string Document)[] Buffers(IReadOnlyList<XmlBytesDocumentOutput> docs,byte[] primary,byte[][] named){ReadOnly(docs);var identities=new HashSet<byte[]>(ReferenceEqualityComparer.Instance);_=identities.Add(primary);foreach(var n in named)_=identities.Add(n);foreach(var d in docs)Require(identities.Add(d.Document),"all output buffers distinct and owned independently of input buffers");return docs.Select(d=>(d.Path,Utf8.GetString(d.Document))).ToArray();}
 static (string Path,string Document)[] Call(int api,byte[] primary,JsonElement[] pairs,byte[][] named,FerruleExecutionContext execution)
 {
  if(api<2){var inputs=pairs.Select((p,i)=>new NamedXmlInput(p.GetProperty("name").GetString()!,Utf8.GetString(named[i]))).ToArray();var text=Utf8.GetString(primary);return Strings(api==0?GeneratedMapping.ExecuteXmlDocumentsWithSources(text,inputs):GeneratedMapping.ExecuteXmlDocumentsWithSources(text,inputs,execution));}
  var bytes=pairs.Select((p,i)=>new NamedXmlBytesInput(p.GetProperty("name").GetString()!,named[i])).ToArray();return Buffers(api==2?GeneratedMapping.ExecuteXmlBytesDocumentsWithSources(primary,bytes):GeneratedMapping.ExecuteXmlBytesDocumentsWithSources(primary,bytes,execution),primary,named);
 }
 static object? Owner(FerruleXmlInputDocumentExecutionException error)=>error.Owner switch

  null=>null,
  FerruleXmlInputDocumentOwner.Input input=>new{kind="Input",source=new{kind=input.Source.Kind.ToString(),index=input.Source.Index,name=input.Source.Name}},
  FerruleXmlInputDocumentOwner.Member member=>Member(member.Output),
  _=>throw new InvalidOperationException("closed source/member owner"),
 };

 static object Boundary(FerruleXmlBoundaryException error)
 {
  var causes=new List<object>();object? runtime=null;for(var cause=error.InnerException;cause is not null;cause=cause.InnerException){causes.Add(new{type=cause.GetType().FullName,detail=cause.Message});Require(causes.Count<32,"bounded original cause chain");if(cause is FerruleRuntimeException r){runtime=r.Error switch{FerruleRuntimeError.EmptyDynamicTargetPath=>new{error=r.Error.ToString(),node=r.Node},FerruleRuntimeError.UnexpectedNamedSource or FerruleRuntimeError.DuplicateNamedSource or FerruleRuntimeError.MissingNamedSource=>new{error=r.Error.ToString(),name=r.Detail},_=>new{error=r.Error.ToString(),detail=r.Detail}};}}
  return new{kind=error.Kind.ToString(),detail=error.Detail,bytes=error.Bytes,limit=error.Limit,causes,runtime};
 }
static object Physical(string xml)
{
    static object Element(XElement element)
    {
        var attributes = element.Attributes().Where(attribute => !attribute.IsNamespaceDeclaration)
            .OrderBy(attribute => attribute.Name.NamespaceName, StringComparer.Ordinal).ThenBy(attribute => attribute.Name.LocalName, StringComparer.Ordinal)
            .Select(attribute => new { name = attribute.Name.LocalName, @namespace = attribute.Name.NamespaceName, value = attribute.Value }).ToArray();
        var children = element.Elements().Select(Element).ToArray();
        var rawText = string.Concat(element.Nodes().OfType<XText>().Select(text => text.Value));
        string? text = rawText;
        if (children.Length != 0)
        {
            Require(rawText.All(character => character is ' ' or '\t' or '\r' or '\n'), "only element-only formatting whitespace");
            text = null;
        }
        return new { name = element.Name.LocalName, @namespace = element.Name.NamespaceName, text, attributes, children };
    }
    return Element(XDocument.Parse(xml, LoadOptions.PreserveWhitespace).Root ?? throw new InvalidOperationException("document root"));
}
static object Snapshot(FerruleInstance input) => input switch
{
    FerruleScalar scalar => Scalar(scalar.Value),
    FerruleGroup group => new { kind = "Group", xml_type_origin = Origin(group.XmlTypeOrigin), fields = group.Fields.Select(field => new { name = field.Name, instance = Snapshot(field.Value) }).ToArray() },
    FerruleRepeated repeated => new { kind = "Repeated", items = repeated.Items.Select(Snapshot).ToArray() },
    FerruleMappedSequence mapped => new { kind = "MappedSequence", items = mapped.Items.Select(Snapshot).ToArray() },
    _ => throw new InvalidOperationException("unexpected closed instance variant"),
};
static object Origin(FerruleXmlTypeOrigin origin) => origin.Kind switch
{
    FerruleXmlTypeOriginKind.Unknown => new { kind = "Unknown" },
    FerruleXmlTypeOriginKind.Absent => new { kind = "Absent" },
    FerruleXmlTypeOriginKind.Explicit => new { kind = "Explicit", identity = origin.Identity },
    FerruleXmlTypeOriginKind.ExplicitPadded => new { kind = "ExplicitPadded", literal = origin.Literal, resolved_identity = origin.Identity },
    _ => throw new InvalidOperationException("unexpected XML type origin"),
};
static object Scalar(FerruleValue value) => value.Kind switch
{
    FerruleValueKind.Null => new { kind = "Scalar", tag = "Null" },
    FerruleValueKind.JsonNull => new { kind = "Scalar", tag = "JsonNull" },
    FerruleValueKind.XmlNil => new { kind = "Scalar", tag = "XmlNil" },
    FerruleValueKind.Bool => new { kind = "Scalar", tag = "Bool", value = value.BooleanValue },
    FerruleValueKind.Int64 => new { kind = "Scalar", tag = "Int", value = value.Int64Value },
    FerruleValueKind.Double => new { kind = "Scalar", tag = "Float", bits_hex = BitConverter.DoubleToUInt64Bits(value.DoubleValue).ToString("x16", CultureInfo.InvariantCulture) },
    FerruleValueKind.String => new { kind = "Scalar", tag = "String", value = value.StringValue },
    _ => throw new InvalidOperationException("unexpected scalar tag"),
};
