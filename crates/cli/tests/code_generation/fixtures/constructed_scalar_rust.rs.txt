use codegen_runtime::{ExecutionContext, FunctionError, JsonBoundaryError, RuntimeError, RuntimeParameters, Value};
use ferrule_generated_mapping::{execute_json, execute_json_bytes, execute_json_bytes_with_context, execute_json_with_context};
use std::path::Path;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let cases: Vec<serde_json::Value> = serde_json::from_str(include_str!("../cases.json"))?;
    for case in &cases {
        let input = case["input"].as_str().unwrap();
        let mut parameters = RuntimeParameters::new();
        for (name, value) in case.get("parameters").and_then(serde_json::Value::as_object).into_iter().flatten() {
            parameters.insert(name, Value::String(value.as_str().unwrap().to_string()))?;
        }
        let context = ExecutionContext::new(Path::new("constructed.ferrule.json")).with_parameters(&parameters);
        verify(execute_json_with_context(input, &context), case);
        verify(execute_json_bytes_with_context(input.as_bytes(), &context).map(|bytes| String::from_utf8(bytes).unwrap()), case);
        if case.get("parameters").is_none() {
            verify(execute_json(input), case);
            verify(execute_json_bytes(input.as_bytes()).map(|bytes| String::from_utf8(bytes).unwrap()), case);
        }
    }
    println!("verified {} constructed scalar cases", cases.len());
    Ok(())
}

fn verify(result: Result<String, JsonBoundaryError>, case: &serde_json::Value) {
    if case["reject"] == true {
        assert!(matches!(result, Err(JsonBoundaryError::InvalidInput { .. })), "{case}: {result:?}");
    } else if let Some(expected) = case.get("execution_error") {
        assert!(matches!(&result, Err(JsonBoundaryError::Execution(RuntimeError::Function(FunctionError::InvalidArgument { function, message })))
            if *function == expected["function"].as_str().unwrap() && *message == expected["message"].as_str().unwrap()), "{case}: {result:?}");
    } else if let Some(expected) = case.get("type_error") {
        assert!(matches!(&result, Err(JsonBoundaryError::Execution(RuntimeError::Function(FunctionError::TypeMismatch { function, got })))
            if *function == expected["function"].as_str().unwrap() && *got == expected["found"].as_str().unwrap()), "{case}: {result:?}");
    } else if let Some(expected) = case.get("boolean_error") {
        assert!(matches!(&result, Err(JsonBoundaryError::Execution(RuntimeError::NotABool { node, found }))
            if *node == expected["node"].as_u64().unwrap() as u32 && *found == expected["found"].as_str().unwrap()), "{case}: {result:?}");
    } else if let Some(expected) = case.get("missing_parameter") {
        assert!(matches!(&result, Err(JsonBoundaryError::Execution(RuntimeError::MissingRuntimeParameter { node, name }))
            if *node == expected["node"].as_u64().unwrap() as u32 && name == expected["name"].as_str().unwrap()), "{case}: {result:?}");
    } else {
        let actual: serde_json::Value = serde_json::from_str(&result.unwrap()).unwrap();
        assert_eq!(actual, case["expected"], "{case}");
    }
}
