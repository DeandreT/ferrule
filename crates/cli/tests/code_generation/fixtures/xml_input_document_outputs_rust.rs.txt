use codegen_runtime::{ExecutionContext,Instance,RuntimeError,RuntimeValue,XmlBoundaryError,XmlInputDocumentOutputsExecutionError,XmlInputDocumentOutputsOwner,XmlInputSource,XmlDocumentOutputsOwner};
use ferrule_generated_mapping as generated;
use serde_json::{Value as Json,json};
use std::{error::Error,path::PathBuf};
#[path="structured_xml_snapshot.rs.txt"] mod snapshot;
fn physical(xml:&str)->Json {
 fn element(node:roxmltree::Node<'_,'_>)->Json {let mut attributes=node.attributes().map(|a|(a.namespace().unwrap_or(""),a.name(),a.value())).collect::<Vec<_>>();attributes.sort();let children=node.children().filter(|n|n.is_element()).map(element).collect::<Vec<_>>();let raw=node.children().filter(|n|n.is_text()).filter_map(|n|n.text()).collect::<String>();let text=if children.is_empty(){Some(raw)}else{assert!(raw.chars().all(|c|matches!(c,' '| '\t'|'\r'|'\n')));None};json!({"name":node.tag_name().name(),"namespace":node.tag_name().namespace().unwrap_or(""),"text":text,"attributes":attributes.iter().map(|(namespace,name,value)|json!({"name":name,"namespace":namespace,"value":value})).collect::<Vec<_>>(),"children":children})}
 element(roxmltree::Document::parse(xml).unwrap().root_element())
}
fn members(value:&Instance)->Json {let Instance::DocumentSet(docs)=value else{panic!("complete named DocumentSet")};json!(docs.iter().map(|d|json!({"path":d.path(),"instance":snapshot::snapshot(d.value())})).collect::<Vec<_>>())}
fn boundary(error:&XmlBoundaryError)->Json {let mut causes=Vec::new();let mut next=error.source();let mut runtime=None;while let Some(cause)=next {causes.push(json!({"detail":cause.to_string(),"debug":format!("{cause:?}")}));assert!(causes.len()<32);if let Some(r)=cause.downcast_ref::<RuntimeError>() {runtime=Some(match r {RuntimeError::EmptyDynamicTargetPath{node}=>json!({"error":"EmptyDynamicTargetPath","node":node}),RuntimeError::UnexpectedNamedSource{name}=>json!({"error":"UnexpectedNamedSource","name":name}),RuntimeError::DuplicateNamedSource{name}=>json!({"error":"DuplicateNamedSource","name":name}),RuntimeError::MissingNamedSource{name}=>json!({"error":"MissingNamedSource","name":name}),RuntimeError::MissingRuntimeValue{value}=>json!({"error":"MissingRuntimeValue","runtime_value":format!("{value:?}")}),_=>json!({"error":"other","debug":format!("{r:?}")})});}next=cause.source();}let resource=error.source().and_then(|c|c.downcast_ref::<codegen_runtime::XmlInputSetResourceError>()).map(|r|json!({"resource":r.resource,"observed_count":r.observed_count,"limit":r.limit}));json!({"kind":format!("{:?}",error.kind),"detail":error.detail,"bytes":error.bytes,"limit":error.limit,"causes":causes,"runtime":runtime,"resource":resource})}
fn owner(error:&XmlInputDocumentOutputsExecutionError)->Json {match &error.owner {None=>Json::Null,Some(XmlInputDocumentOutputsOwner::Input(source))=>json!({"kind":"Input","source":match source {XmlInputSource::Primary=>json!({"kind":"Primary","index":null,"name":null}),XmlInputSource::Named{index,name}=>json!({"kind":"Named","index":index,"name":name})}}),Some(XmlInputDocumentOutputsOwner::Output(output))=>json!({"kind":"Output","owner":match output {XmlDocumentOutputsOwner::Primary=>json!({"kind":"Primary"}),XmlDocumentOutputsOwner::NamedMember{declaration_index,name,index,path}=>json!({"kind":"NamedMember","declaration_index":declaration_index,"name":name,"index":index,"path":path})}}),Some(_)=>panic!("future phase owner")}}
type Mixed=(String,Vec<(String,Vec<(String,String)>)>);
fn call(api:usize,primary:&[u8],pairs:&[Json],documents:&[Vec<u8>],execution:&ExecutionContext<'_>)->Result<Mixed,XmlInputDocumentOutputsExecutionError> {
 fn strings(output:generated::XmlDocumentExecutionOutputs)->Mixed {(output.primary,output.extras.into_iter().map(|e|(e.name.to_owned(),e.documents.into_iter().map(|d|(d.path,d.document)).collect())).collect())}
 fn buffers(output:generated::XmlBytesDocumentExecutionOutputs,primary:&[u8],documents:&[Vec<u8>])->Mixed {let mut live=std::collections::HashSet::new();live.insert(primary.as_ptr());for d in documents{live.insert(d.as_ptr());}assert!(live.insert(output.primary.as_ptr()));for e in &output.extras{for d in &e.documents{assert!(live.insert(d.document.as_ptr()),"all live original returned buffers distinct and owned");}}(String::from_utf8(output.primary).unwrap(),output.extras.into_iter().map(|e|(e.name.to_owned(),e.documents.into_iter().map(|d|(d.path,String::from_utf8(d.document).unwrap())).collect())).collect())}
 if api<2 {let inputs=pairs.iter().zip(documents).map(|(p,d)|generated::NamedXmlInput{name:p["name"].as_str().unwrap(),document:std::str::from_utf8(d).unwrap()}).collect::<Vec<_>>();let text=std::str::from_utf8(primary).unwrap();let output=if api==0{generated::execute_xml_document_outputs_with_sources(text,&inputs)}else{generated::execute_xml_document_outputs_with_sources_and_context(text,&inputs,execution)};output.map(strings)}else{let inputs=pairs.iter().zip(documents).map(|(p,d)|generated::NamedXmlBytesInput{name:p["name"].as_str().unwrap(),document:d}).collect::<Vec<_>>();let output=if api==2{generated::execute_xml_bytes_document_outputs_with_sources(primary,&inputs)}else{assert_eq!(api,3);generated::execute_xml_bytes_document_outputs_with_sources_and_context(primary,&inputs,execution)};output.map(|o|buffers(o,primary,documents))}
}
fn main()->Result<(),Box<dyn Error>> {
 let directory=PathBuf::from(std::env::args().nth(1).expect("fixture directory"));let cases:Json=serde_json::from_slice(&std::fs::read(directory.join("cases.json"))?)?;let descriptors:Json=serde_json::from_slice(&std::fs::read(directory.join("descriptors.json"))?)?;let project_path=directory.join("project.json");let execution=ExecutionContext::new(&project_path).with_current_datetime("2026-10-04T00:00:00Z");let source_schema=descriptors["source"].as_str().unwrap();let target_schema=descriptors["target"].as_str().unwrap();let hints=descriptors["hints"].as_str().unwrap();let names=["z-rates","a-count","unused-padding"];let schemas=descriptors["inputs"].as_array().unwrap();let targets=descriptors["outputs"].as_array().unwrap();assert_eq!(schemas.len(),3);assert_eq!(targets.len(),2);
 for test in cases.as_array().unwrap() {
  let name=test["case"].as_str().unwrap();let primary=std::fs::read(directory.join(test["primary"].as_str().unwrap()))?;let pairs=test["inputs"].as_array().unwrap();let documents=pairs.iter().map(|p|std::fs::read(directory.join(p["file"].as_str().unwrap()))).collect::<Result<Vec<_>,_>>()?;let phase=test["primitive"]["phase"].as_str();let mut original:Option<Box<XmlBoundaryError>>=None;let mut missing:Option<Box<XmlBoundaryError>>=None;
  if test["semantic"]==true {
   let parsed=codegen_runtime::parse_structured_xml_bytes(source_schema,&primary)?;let declared=names.iter().enumerate().map(|(i,n)|{let supplied=pairs.iter().position(|p|p["name"]==*n).unwrap();codegen_runtime::parse_structured_xml_bytes(schemas[i]["schema"].as_str().unwrap(),&documents[supplied])}).collect::<Result<Vec<_>,_>>()?;let typed=(0..3).rev().map(|i|codegen_runtime::NamedInput{name:names[i],instance:&declared[i]}).collect::<Vec<_>>();let input_records=names.iter().zip(&declared).map(|(name,value)|json!({"name":name,"instance":snapshot::snapshot(value)})).collect::<Vec<_>>();
   if test["no_context_missing"]==true {let error=generated::execute_outputs_with_sources(&parsed,&typed).unwrap_err();assert!(matches!(error,RuntimeError::MissingRuntimeValue{value:RuntimeValue::CurrentDateTime}));missing=Some(Box::new(XmlBoundaryError::from(error)));}
   match generated::execute_outputs_with_sources_and_context(&parsed,&typed,&execution) {
    Ok(mapped)=>{assert_eq!(mapped.extras.len(),2);for (e,t) in mapped.extras.iter().zip(targets){assert_eq!(e.name,t["name"].as_str().unwrap());}println!("{}",json!({"case":name,"api":-1,"source":snapshot::snapshot(&parsed),"inputs":input_records,"primary":snapshot::snapshot(&mapped.primary),"extras":mapped.extras.iter().map(|e|json!({"name":e.name,"members":members(&e.instance)})).collect::<Vec<_>>(),"mapping_failure":false}));
     if phase==Some("output") {let target=test["primitive"]["target"].as_str().unwrap();original=Some(Box::new(if target=="primary"{codegen_runtime::serialize_xml_document(target_schema,&mapped.primary,true,true,None,Some(hints)).unwrap_err()}else{let declaration=targets.iter().position(|t|t["name"]==target).unwrap();let Instance::DocumentSet(docs)=&mapped.extras[declaration].instance else{panic!("typed named documents")};let i=test["primitive"]["index"].as_u64().unwrap() as usize;codegen_runtime::serialize_xml_document(targets[declaration]["schema"].as_str().unwrap(),docs[i].value(),true,true,None,Some(targets[declaration]["hints"].as_str().unwrap())).unwrap_err()}));}
    }
    Err(error)=>{assert_eq!(phase,Some("mapping"));assert!(matches!(error,RuntimeError::EmptyDynamicTargetPath{node:3}));original=Some(Box::new(XmlBoundaryError::from(error)));println!("{}",json!({"case":name,"api":-1,"source":snapshot::snapshot(&parsed),"inputs":input_records,"primary":null,"extras":null,"mapping_failure":true}));}
   }
  }
  if phase==Some("parser") {let input=test["primitive"]["source"].as_str().unwrap();let (schema,bytes)=if input=="primary"{(source_schema,primary.as_slice())}else{let i=names.iter().position(|n|*n==input).unwrap();let supplied=pairs.iter().position(|p|p["name"]==input).unwrap();(schemas[i]["schema"].as_str().unwrap(),documents[supplied].as_slice())};original=Some(Box::new(codegen_runtime::parse_structured_xml_bytes(schema,bytes).unwrap_err()));}
  if phase==Some("names") {let supplied=pairs.iter().map(|p|p["name"].as_str().unwrap()).collect::<Vec<_>>();let e=codegen_runtime::xml_input_indices(&names,&supplied).unwrap_err();assert!(e.input.is_none());original=Some(e.boundary);}
  if phase==Some("count") {let e=codegen_runtime::XmlInputSetBudget::new(pairs.len()+1).err().unwrap();assert!(e.input.is_none());original=Some(e.boundary);}
  for api in test["apis"].as_array().unwrap(){let api=api.as_u64().unwrap() as usize;let no_context=test["no_context_missing"]==true&&api%2==0;let expected_kind=if no_context{"Mapping"}else{test["kind"].as_str().unwrap()};
   match call(api,&primary,pairs,&documents,&execution) {
    Ok((primary,extras))=>{assert_eq!(expected_kind,"ok");assert_eq!(extras.len(),2);let actual=physical(&primary);assert_eq!(actual,physical(&std::fs::read_to_string(directory.join(format!("native-{name}-primary.xml")))?));let output=extras.into_iter().enumerate().map(|(decl,(target,docs))|{assert_eq!(target,targets[decl]["name"]);json!({"name":target,"documents":docs.into_iter().enumerate().map(|(i,(path,document))|{let actual=physical(&document);let native=physical(&std::fs::read_to_string(directory.join(format!("native-{name}-{decl}-{i}.xml"))).unwrap());assert_eq!(actual,native);json!({"path":path,"document":document,"physical":actual,"native_physical_match":true})}).collect::<Vec<_>>()})}).collect::<Vec<_>>();println!("{}",json!({"case":name,"api":api,"kind":"ok","primary":{"document":primary,"physical":actual,"native_physical_match":true},"extras":output,"owned_byte_buffers":if api>=2{Some(true)}else{None}}));}
    Err(e)=>{assert_eq!(format!("{:?}",e.boundary.kind),expected_kind);let identity=e.source().and_then(|c|c.downcast_ref::<XmlBoundaryError>()).is_some_and(|b|std::ptr::eq(b,e.boundary.as_ref()));assert!(identity);let wanted=if no_context{missing.as_ref()}else{original.as_ref()}.expect("complete independent own primitive refusal");assert_eq!(boundary(&e.boundary),boundary(wanted));println!("{}",json!({"case":name,"api":api,"kind":expected_kind,"wrapper":"XmlInputDocumentOutputsExecutionError","owner":owner(&e),"boundary":boundary(&e.boundary),"original_boundary":boundary(wanted),"boundary_identity":identity,"original_boundary_match":true,"no_partial_result":true}));}
   }
  }
 }
 Ok(())
}
