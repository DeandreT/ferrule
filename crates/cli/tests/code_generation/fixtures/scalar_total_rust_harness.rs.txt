use std::path::Path;
use codegen_runtime::{ExecutionContext, FunctionError, Instance, JsonBoundaryError, RuntimeError, Value};
use ferrule_generated_mapping as mapping;
use serde_json::{Value as Json, json};

fn runtime_error(error: RuntimeError) -> Json {
    let detail=error.to_string();
    match error {
        RuntimeError::Function(FunctionError::TypeMismatch { function, got }) =>
            json!({"kind":"error","category":"FunctionType","function":function,"found":got,"detail":detail}),
        other => json!({"kind":"error","category":"OtherRuntime","debug":format!("{other:?}"),"detail":detail}),
    }
}
fn boundary_error(error: JsonBoundaryError) -> Json {
    match error {
        JsonBoundaryError::Execution(inner) => runtime_error(inner),
        other => json!({"kind":"error","category":"JsonBoundary","debug":format!("{other:?}"),"detail":other.to_string()}),
    }
}
fn text(result: Result<String, JsonBoundaryError>) -> Json {
    match result { Ok(document) => json!({"kind":"ok","json":document}), Err(error)=>boundary_error(error) }
}
fn bytes(result: Result<Vec<u8>, JsonBoundaryError>) -> Json {
    match result { Ok(document) => json!({"kind":"ok","json":String::from_utf8(document).expect("generated bytes are UTF8")}), Err(error)=>boundary_error(error) }
}
fn typed_value(value: Instance, schema:&str) -> Json {
    let (tag, integer, float_bits)=match value.field("total") {
        Some(Instance::Scalar(Value::Int(value)))=>("Int",Some(*value),None),
        Some(Instance::Scalar(Value::Float(value)))=>("Float",None,Some(format!("{:016x}",value.to_bits()))),
        Some(other)=>return json!({"kind":"unexpected-target-tag","value":format!("{other:?}")}),
        None=>return json!({"kind":"missing-total"}),
    };
    match codegen_runtime::serialize_json(schema,&value) {
        Ok(document)=>json!({"kind":"ok","json":document,"target_tag":tag,"target_int":integer,"target_f64_bits":float_bits}),
        Err(error)=>boundary_error(error),
    }
}
fn typed(result: Result<Instance, RuntimeError>, schema:&str) -> Json {
    match result {Ok(value)=>typed_value(value,schema),Err(error)=>runtime_error(error)}
}
fn interpreter(project:&mapping_ir::Project, source:&Instance, schema:&str) -> Json {
    match engine::run(project,source) {
        Ok(value)=>typed_value(value,schema),
        Err(engine::EngineError::Function(error))=>runtime_error(RuntimeError::Function(error)),
        Err(error)=>json!({"kind":"error","category":"OtherEngine","detail":error.to_string()}),
    }
}
fn scalar_tag(value:Option<&Instance>)->&'static str {
    match value {Some(Instance::Scalar(Value::Int(_)))=>"Int",Some(Instance::Scalar(Value::Float(_)))=>"Float",Some(Instance::Scalar(Value::Null))=>"Null",None=>"Missing",_=>"Other"}
}
fn quantity(value:Option<&Instance>)->Option<i64> {match value {Some(Instance::Scalar(Value::Int(value)))=>Some(*value),_=>None}}
fn price_bits(value:Option<&Instance>)->Option<String> {match value {Some(Instance::Scalar(Value::Float(value)))=>Some(format!("{:016x}",value.to_bits())),_=>None}}
fn consumed(source:&Instance)->Json {
    let items=source.field("Items").and_then(|items|items.field("item"));
    match items {
        Some(Instance::Repeated(rows))=>json!({"count":rows.len(),"fields":rows.iter().map(|row|json!({"quantity_tag":scalar_tag(row.field("quantity")),"quantity_value":quantity(row.field("quantity")),"price_tag":scalar_tag(row.field("price")),"price_f64_bits":price_bits(row.field("price"))})).collect::<Vec<_>>()}),
        other=>json!({"unexpected_collection":format!("{other:?}")}),
    }
}
fn emit(case:&str,api:&str,result:Json) { println!("{}",json!({"case":case,"api":api,"result":result})); }
fn main()->Result<(),Box<dyn std::error::Error>> {
    let args=std::env::args().skip(1).collect::<Vec<_>>();
    assert_eq!(args.len(),6,"case JSON XML source-schema target-schema project-path");
    let case=&args[0];let input=std::fs::read_to_string(&args[1])?;let source_schema=std::fs::read_to_string(&args[3])?;let target_schema=std::fs::read_to_string(&args[4])?;
    let project=mapping_ir::project_file::decode_str(&std::fs::read_to_string(&args[5])?)?;
    let context=ExecutionContext::new(Path::new(&args[5])).with_current_datetime("2026-10-03T00:00:00Z");
    emit(case,"json",text(mapping::execute_json(&input)));
    emit(case,"json-context",text(mapping::execute_json_with_context(&input,&context)));
    emit(case,"bytes",bytes(mapping::execute_json_bytes(input.as_bytes())));
    emit(case,"bytes-context",bytes(mapping::execute_json_bytes_with_context(input.as_bytes(),&context)));
    match codegen_runtime::parse_json(&source_schema,&input) {
        Ok(source)=> {
            emit(case,"consumed-json-fields",consumed(&source));
            emit(case,"interpreter-json",interpreter(&project,&source,&target_schema));
            emit(case,"typed-json",typed(mapping::execute(&source),&target_schema));
            emit(case,"typed-json-context",typed(mapping::execute_with_context(&source,&context),&target_schema));
        },
        Err(error)=>emit(case,"typed-json-source-parse",boundary_error(error)),
    }
    let source_schema:ir::SchemaNode=serde_json::from_str(&source_schema)?;
    match format_xml::read(Path::new(&args[2]),&source_schema) {
        Ok(source)=> {
            emit(case,"consumed-native-XML-fields",consumed(&source));
            emit(case,"interpreter-native-XML",interpreter(&project,&source,&target_schema));
            emit(case,"typed-native-XML",typed(mapping::execute(&source),&target_schema));
            emit(case,"typed-native-XML-context",typed(mapping::execute_with_context(&source,&context),&target_schema));
        },
        Err(error)=>emit(case,"native-XML-source-parse",json!({"kind":"error","detail":error.to_string()})),
    }
    Ok(())
}
