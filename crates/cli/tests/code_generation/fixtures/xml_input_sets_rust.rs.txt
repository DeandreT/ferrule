use codegen_runtime::{
    ExecutionContext, RuntimeError, XmlBoundaryError, XmlExecutionError, XmlInputSource,
    XmlOutputSetError, XmlOutputTarget,
};
use ferrule_generated_mapping as generated;
use serde_json::{Value as Json, json};
use std::error::Error;
use std::path::Path;

#[path = "structured_xml_snapshot.rs.txt"]
mod snapshot;

fn physical(xml: &str) -> Json {
    fn element(node: roxmltree::Node<'_, '_>) -> Json {
        let mut attributes = node
            .attributes()
            .map(|attribute| {
                (
                    attribute.namespace().unwrap_or(""),
                    attribute.name(),
                    attribute.value(),
                )
            })
            .collect::<Vec<_>>();
        attributes.sort_by_key(|(namespace, name, _)| (*namespace, *name));
        let children = node
            .children()
            .filter(|child| child.is_element())
            .map(element)
            .collect::<Vec<_>>();
        let text = if children.is_empty() {
            Some(
                node.children()
                    .filter(|child| child.is_text())
                    .filter_map(|child| child.text())
                    .collect::<String>(),
            )
        } else {
            None
        };
        json!({"name":node.tag_name().name(),"namespace":node.tag_name().namespace().unwrap_or(""),"text":text,
            "attributes":attributes.iter().map(|(namespace,name,value)|json!({"name":name,"namespace":namespace,"value":value})).collect::<Vec<_>>(),"children":children})
    }
    element(
        roxmltree::Document::parse(xml)
            .expect("complete returned XML must parse")
            .root_element(),
    )
}
fn document(xml: String) -> Json {
    json!({"physical":physical(&xml),"xml":xml})
}
fn boundary(error: &XmlBoundaryError) -> Json {
    let mut function = None;
    let mut found = None;
    let mut runtime_value = None;
    let mut inner_sources = Vec::new();
    let mut resource = None;
    let mut writer_error = None;
    let mut source = error.source();
    while let Some(current) = source {
        if let Some(cause) = current.downcast_ref::<format_xml::XmlFormatError>() {
            writer_error = Some(json!({"type":"XmlFormatError","debug":format!("{cause:?}")}));
        }
        inner_sources.push(json!({"detail":current.to_string(),"debug":format!("{current:?}")}));
        if let Some(cause) = current.downcast_ref::<codegen_runtime::XmlOutputSetResourceError>() {
            resource = Some(
                json!({"resource":cause.resource,"observed_count":cause.observed_count,"limit":cause.limit}),
            );
        }
        if let Some(RuntimeError::Function(codegen_runtime::FunctionError::TypeMismatch {
            function: name,
            got,
        })) = current.downcast_ref::<RuntimeError>()
        {
            function = Some(*name);
            found = Some(*got);
        }
        if let Some(RuntimeError::MissingRuntimeValue { value }) =
            current.downcast_ref::<RuntimeError>()
        {
            runtime_value = Some(format!("{value:?}"));
        }
        if let Some(cause) = current.downcast_ref::<codegen_runtime::XmlInputSetResourceError>() {
            resource = Some(
                json!({"resource":cause.resource,"observed_count":cause.observed_count,"limit":cause.limit}),
            );
        }
        source = current.source();
    }
    let name_error = error
        .source()
        .and_then(|source| source.downcast_ref::<RuntimeError>())
        .and_then(|cause| match cause {
            RuntimeError::MissingNamedSource { name } => {
                Some(json!({"kind":"MissingNamedSource","name":name}))
            }
            RuntimeError::DuplicateNamedSource { name } => {
                Some(json!({"kind":"DuplicateNamedSource","name":name}))
            }
            RuntimeError::UnexpectedNamedSource { name } => {
                Some(json!({"kind":"UnexpectedNamedSource","name":name}))
            }
            _ => None,
        });
    json!({"kind":format!("{:?}",error.kind),"detail":error.detail,"raw_debug":format!("{error:?}"),"function":function,"found":found,"runtime_value":runtime_value,"boundary_bytes":error.bytes,"boundary_limit":error.limit,"inner_sources":inner_sources,"resource":resource,"writer_error":writer_error,"name_error":name_error,"no_output":true})
}
fn outputs(result: Result<generated::XmlExecutionOutputs, XmlOutputSetError>) -> Json {
    match result {
        Ok(output) => {
            json!({"kind":"ok","primary":document(output.primary),"extras":output.extras.into_iter().map(|target|{
            json!({"name":target.name,"physical":physical(&target.document),"xml":target.document})
        }).collect::<Vec<_>>()})
        }
        Err(error) => {
            let mut result = boundary(&error.boundary);
            result["target"] = match error.target {
                None => Json::Null,
                Some(XmlOutputTarget::Primary) => {
                    json!({"kind":"Primary","index":null,"name":null})
                }
                Some(XmlOutputTarget::Named { index, name }) => {
                    json!({"kind":"Named","index":index,"name":name})
                }
            };
            result["boundary_preserved"] = json!(
                error
                    .source()
                    .and_then(|source| source.downcast_ref::<XmlBoundaryError>())
                    .is_some_and(|boundary| std::ptr::eq(boundary, &error.boundary))
            );
            result
        }
    }
}
fn bytes_outputs(result: Result<generated::XmlBytesExecutionOutputs, XmlOutputSetError>) -> Json {
    outputs(result.map(|output| {
        generated::XmlExecutionOutputs {
            primary: String::from_utf8(output.primary).expect("strict UTF8 primary"),
            extras: output
                .extras
                .into_iter()
                .map(|target| generated::NamedXmlOutput {
                    name: target.name,
                    document: String::from_utf8(target.document)
                        .expect("strict UTF8 named document"),
                })
                .collect(),
        }
    }))
}
fn singular(result: Result<String, XmlBoundaryError>) -> Json {
    match result {
        Ok(xml) => json!({"kind":"ok","primary":document(xml)}),
        Err(error) => boundary(&error),
    }
}

fn owned_error(error: XmlExecutionError) -> Json {
    let mut result = boundary(&error.boundary);
    result["input"] = match error.input {
        None => Json::Null,
        Some(XmlInputSource::Primary) => json!({"kind":"Primary","index":null,"name":null}),
        Some(XmlInputSource::Named { index, name }) => {
            json!({"kind":"Named","index":index,"name":name})
        }
    };
    result["output"] = match error.output {
        None => Json::Null,
        Some(XmlOutputTarget::Primary) => json!({"kind":"Primary","index":null,"name":null}),
        Some(XmlOutputTarget::Named { index, name }) => {
            json!({"kind":"Named","index":index,"name":name})
        }
    };
    result["boundary_preserved"] = json!(
        error
            .source()
            .and_then(|source| source.downcast_ref::<XmlBoundaryError>())
            .is_some_and(|boundary| std::ptr::eq(boundary, error.boundary.as_ref()))
    );
    result
}
fn source_outputs(result: Result<generated::XmlExecutionOutputs, XmlExecutionError>) -> Json {
    match result {
        Ok(output) => outputs(Ok(output)),
        Err(error) => owned_error(error),
    }
}
fn source_bytes_outputs(
    result: Result<generated::XmlBytesExecutionOutputs, XmlExecutionError>,
) -> Json {
    source_outputs(result.map(|output| {
        generated::XmlExecutionOutputs {
            primary: String::from_utf8(output.primary).expect("strict UTF8 primary"),
            extras: output
                .extras
                .into_iter()
                .map(|target| generated::NamedXmlOutput {
                    name: target.name,
                    document: String::from_utf8(target.document).expect("strict UTF8 named output"),
                })
                .collect(),
        }
    }))
}
fn source_singular(result: Result<String, XmlExecutionError>) -> Json {
    match result {
        Ok(xml) => json!({"kind":"ok","primary":document(xml)}),
        Err(error) => owned_error(error),
    }
}
fn bytes(value: &Json) -> Vec<u8> {
    value
        .as_array()
        .expect("exact input bytes")
        .iter()
        .map(|value| u8::try_from(value.as_u64().expect("unsigned byte")).expect("byte range"))
        .collect()
}
fn main() -> Result<(), Box<dyn Error>> {
    let args = std::env::args().skip(1).collect::<Vec<_>>();
    assert_eq!(
        args.len(),
        4,
        "cases descriptors exact-project-path variant"
    );
    let cases: Json = serde_json::from_slice(&std::fs::read(&args[0])?)?;
    let descriptors: Json = serde_json::from_slice(&std::fs::read(&args[1])?)?;
    let context =
        ExecutionContext::new(Path::new(&args[2])).with_current_datetime("2026-10-04T00:00:00Z");
    for test in cases.as_array().expect("cases") {
        let case = test["case"].as_str().expect("case");
        let primary = bytes(&test["primary"]);
        let pairs = test["inputs"].as_array().expect("named inputs");
        let named = pairs
            .iter()
            .map(|input| bytes(&input["bytes"]))
            .collect::<Vec<_>>();
        let texts = named
            .iter()
            .map(|input| std::str::from_utf8(input))
            .collect::<Result<Vec<_>, _>>();
        let byte_inputs = pairs
            .iter()
            .zip(&named)
            .map(|(input, document)| generated::NamedXmlBytesInput {
                name: input["name"].as_str().expect("supplied exact name"),
                document,
            })
            .collect::<Vec<_>>();
        let emit = |api: &str, result: Json| {
            println!("{}", json!({"case":case,"api":api,"result":result}))
        };
        if let (Ok(primary_text), Ok(texts)) = (std::str::from_utf8(&primary), texts) {
            let inputs = pairs
                .iter()
                .zip(texts)
                .map(|(input, document)| generated::NamedXmlInput {
                    name: input["name"].as_str().expect("supplied exact name"),
                    document,
                })
                .collect::<Vec<_>>();
            emit(
                "xml-outputs-sources",
                source_outputs(generated::execute_xml_outputs_with_sources(
                    primary_text,
                    &inputs,
                )),
            );
            emit(
                "xml-outputs-sources-context",
                source_outputs(generated::execute_xml_outputs_with_sources_and_context(
                    primary_text,
                    &inputs,
                    &context,
                )),
            );
            emit(
                "xml-sources",
                source_singular(generated::execute_xml_with_sources(primary_text, &inputs)),
            );
            emit(
                "xml-sources-context",
                source_singular(generated::execute_xml_with_sources_and_context(
                    primary_text,
                    &inputs,
                    &context,
                )),
            );
        }
        emit(
            "bytes-outputs-sources",
            source_bytes_outputs(generated::execute_xml_bytes_outputs_with_sources(
                &primary,
                &byte_inputs,
            )),
        );
        emit(
            "bytes-outputs-sources-context",
            source_bytes_outputs(
                generated::execute_xml_bytes_outputs_with_sources_and_context(
                    &primary,
                    &byte_inputs,
                    &context,
                ),
            ),
        );
        emit(
            "bytes-sources",
            source_singular(
                generated::execute_xml_bytes_with_sources(&primary, &byte_inputs)
                    .map(|value| String::from_utf8(value).expect("strict UTF8 primary")),
            ),
        );
        emit(
            "bytes-sources-context",
            source_singular(
                generated::execute_xml_bytes_with_sources_and_context(
                    &primary,
                    &byte_inputs,
                    &context,
                )
                .map(|value| String::from_utf8(value).expect("strict UTF8 primary")),
            ),
        );
        if let Ok(text) = std::str::from_utf8(&primary) {
            emit("xml-outputs", outputs(generated::execute_xml_outputs(text)));
            emit(
                "xml-outputs-context",
                outputs(generated::execute_xml_outputs_with_context(text, &context)),
            );
            emit("xml", singular(generated::execute_xml(text)));
            emit(
                "xml-context",
                singular(generated::execute_xml_with_context(text, &context)),
            );
        }
        emit(
            "bytes-outputs",
            bytes_outputs(generated::execute_xml_bytes_outputs(&primary)),
        );
        emit(
            "bytes-outputs-context",
            bytes_outputs(generated::execute_xml_bytes_outputs_with_context(
                &primary, &context,
            )),
        );
        emit(
            "bytes",
            singular(
                generated::execute_xml_bytes(&primary)
                    .map(|value| String::from_utf8(value).expect("strict UTF8 primary")),
            ),
        );
        emit(
            "bytes-context",
            singular(
                generated::execute_xml_bytes_with_context(&primary, &context)
                    .map(|value| String::from_utf8(value).expect("strict UTF8 primary")),
            ),
        );
        if test["diagnostics"] == false {
            continue;
        }
        let mut parsed = Vec::new();
        for (index, input) in std::iter::once(&primary).chain(named.iter()).enumerate() {
            let owner = if index == 0 {
                "primary"
            } else {
                pairs[index - 1]["schema"]
                    .as_str()
                    .expect("fixture schema owner")
            };
            let descriptor = descriptors[owner].as_str().expect("exact own descriptor");
            let api = if index == 0 {
                "source-primary".to_owned()
            } else {
                format!("source-{}", index - 1)
            };
            if let Ok(text) = std::str::from_utf8(input) {
                match codegen_runtime::parse_structured_xml(descriptor, text) {
                    Ok(instance) => emit(
                        &api,
                        json!({"kind":"ok","snapshot":snapshot::snapshot(&instance)}),
                    ),
                    Err(error) => emit(&api, boundary(&error)),
                }
            }
            match codegen_runtime::parse_structured_xml_bytes(descriptor, input) {
                Ok(instance) => {
                    emit(
                        &format!("{api}-bytes"),
                        json!({"kind":"ok","snapshot":snapshot::snapshot(&instance)}),
                    );
                    parsed.push(Some(instance));
                }
                Err(error) => {
                    emit(&format!("{api}-bytes"), boundary(&error));
                    parsed.push(None);
                }
            }
        }
        for with_context in [false, true] {
            let api = if with_context {
                "typed-outputs-context"
            } else {
                "typed-outputs"
            };
            if parsed.iter().any(Option::is_none) {
                emit(api, json!({"kind":"blocked_input","attempted":false}));
                continue;
            }
            let inputs = pairs
                .iter()
                .zip(parsed.iter().skip(1))
                .map(|(input, instance)| codegen_runtime::NamedInput {
                    name: input["name"].as_str().expect("supplied name"),
                    instance: instance.as_ref().expect("parsed own input"),
                })
                .collect::<Vec<_>>();
            let primary = parsed[0].as_ref().expect("parsed primary");
            let result = if with_context {
                generated::execute_outputs_with_sources_and_context(primary, &inputs, &context)
            } else {
                generated::execute_outputs_with_sources(primary, &inputs)
            };
            match result {
                Ok(output) => emit(
                    api,
                    json!({"kind":"ok","primary":snapshot::snapshot(&output.primary),"extras":output.extras.iter().map(|target|json!({"name":target.name,"instance":snapshot::snapshot(&target.instance)})).collect::<Vec<_>>()}),
                ),
                Err(error) => emit(api, boundary(&XmlBoundaryError::from(error))),
            }
        }
    }
    Ok(())
}
