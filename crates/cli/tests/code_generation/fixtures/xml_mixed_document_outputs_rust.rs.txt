use codegen_runtime::{ExecutionContext, Instance, RuntimeError, XmlBoundaryError, XmlBoundaryErrorKind, XmlDocumentOutputsExecutionError, XmlDocumentOutputsOwner};
use ferrule_generated_mapping as generated;
use serde_json::{Value as Json, json};
use std::error::Error;
use std::path::{Path, PathBuf};

#[path = "structured_xml_snapshot.rs.txt"]
mod snapshot;

fn members(value: &Instance) -> Json {
    let Instance::DocumentSet(documents) = value else { panic!("complete document set") };
    json!(documents.iter().map(|document| json!({"path":document.path(),"instance":snapshot::snapshot(document.value())})).collect::<Vec<_>>())
}
fn physical(xml: &str) -> Json {
    fn element(node: roxmltree::Node<'_, '_>) -> Json {
        let mut attributes = node.attributes().map(|attribute|
            (attribute.namespace().unwrap_or(""), attribute.name(), attribute.value())).collect::<Vec<_>>();
        attributes.sort_by_key(|(namespace, name, _)| (*namespace, *name));
        let children = node.children().filter(|child| child.is_element()).map(element).collect::<Vec<_>>();
        let raw_text = node.children().filter(|child| child.is_text()).filter_map(|child| child.text()).collect::<String>();
        let text = if children.is_empty() { Some(raw_text) } else {
            assert!(raw_text.chars().all(|character| matches!(character, ' ' | '\t' | '\r' | '\n')), "only element-only formatting whitespace");
            None
        };
        json!({"name":node.tag_name().name(),"namespace":node.tag_name().namespace().unwrap_or(""),"text":text,
            "attributes":attributes.iter().map(|(namespace,name,value)| json!({"name":name,"namespace":namespace,"value":value})).collect::<Vec<_>>(),"children":children})
    }
    element(roxmltree::Document::parse(xml).expect("complete well-formed output").root_element())
}
fn boundary(error: &XmlBoundaryError) -> Json {
    let mut causes = Vec::new();
    let mut current = error.source();
    while let Some(cause) = current {
        causes.push(json!({"detail":cause.to_string(),"debug":format!("{cause:?}")}));
        current = cause.source();
    }
    json!({"kind":format!("{:?}",error.kind),"detail":error.detail,"bytes":error.bytes,"limit":error.limit,"causes":causes})
}
type Mixed = (String, Vec<(String, Vec<(String, String)>)>);
fn call(api: usize, text: &str, bytes: &[u8], execution: &ExecutionContext<'_>) -> Result<Mixed, XmlDocumentOutputsExecutionError> {
    fn strings(output: generated::XmlDocumentExecutionOutputs) -> Mixed {
        (output.primary, output.extras.into_iter().map(|extra| (extra.name.to_owned(), extra.documents.into_iter().map(|document| (document.path, document.document)).collect())).collect())
    }
    fn buffers(output: generated::XmlBytesDocumentExecutionOutputs) -> Mixed {
        (String::from_utf8(output.primary).expect("owned primary UTF-8"), output.extras.into_iter().map(|extra| (extra.name.to_owned(), extra.documents.into_iter().map(|document| (document.path, String::from_utf8(document.document).expect("owned member UTF-8"))).collect())).collect())
    }
    match api {
        0 => generated::execute_xml_document_outputs(text).map(strings),
        1 => generated::execute_xml_document_outputs_with_context(text, execution).map(strings),
        2 => generated::execute_xml_bytes_document_outputs(bytes).map(buffers),
        3 => generated::execute_xml_bytes_document_outputs_with_context(bytes, execution).map(buffers),
        _ => panic!("one of four mixed APIs"),
    }
}
fn main() -> Result<(), Box<dyn Error>> {
    let directory = PathBuf::from(std::env::args().nth(1).expect("fixture directory"));
    let source_schema = std::fs::read_to_string(directory.join("source-schema.txt"))?;
    let primary_schema = std::fs::read_to_string(directory.join("primary-schema.txt"))?;
    let audit_schema = std::fs::read_to_string(directory.join("audit-schema.txt"))?;
    let execution = ExecutionContext::new(Path::new("fixture.json"));
    for case in ["ordered", "empty", "selected", "late-output", "primary-output", "mapping-primary", "mapping-named"] {
        let bytes = std::fs::read(directory.join(format!("{case}.xml")))?;
        let text = std::str::from_utf8(&bytes)?;
        let parsed = codegen_runtime::parse_structured_xml_bytes(&source_schema, &bytes)?;
        let mapped = generated::execute_outputs_with_context(&parsed, &execution);
        let mut original_output = None;
        let mut original_mapping = None;
        match mapped {
            Ok(mapped) => {
                assert_eq!(mapped.extras.len(), 1);
                assert_eq!(mapped.extras[0].name, "Audit");
                println!("{}", json!({"case":case,"api":-1,"source":snapshot::snapshot(&parsed),"primary":snapshot::snapshot(&mapped.primary),"members":members(&mapped.extras[0].instance),"mapping_failure":false}));
                if case == "late-output" {
                    let Instance::DocumentSet(documents) = &mapped.extras[0].instance else { panic!("named document set") };
                    assert_eq!((documents.len(), documents[1].path()), (2, "bad.xml"));
                    original_output = Some(codegen_runtime::serialize_xml_document(&audit_schema, documents[1].value(), true, true, None, None).unwrap_err());
                } else if case == "primary-output" {
                    original_output = Some(codegen_runtime::serialize_xml_document(&primary_schema, &mapped.primary, true, true, None, None).unwrap_err());
                }
            }
            Err(error) => {
                assert!(case.starts_with("mapping-"));
                assert!(matches!(error, RuntimeError::EmptyDynamicTargetPath { node: 0 }));
                original_mapping = Some(error.to_string());
                println!("{}", json!({"case":case,"api":-1,"source":snapshot::snapshot(&parsed),"primary":null,"members":null,"mapping_failure":true}));
            }
        }
        for api in 0..4 {
            match call(api, text, &bytes, &execution) {
                Ok((primary, extras)) => {
                    assert!(matches!(case, "ordered" | "empty" | "selected"));
                    assert_eq!(extras.len(), 1);
                    assert_eq!(extras[0].0, "Audit");
                    let primary_physical = physical(&primary);
                    assert_eq!(primary_physical, physical(&std::fs::read_to_string(directory.join(format!("native-{case}-primary.xml")))?));
                    let extras = extras.into_iter().map(|(name, documents)| {
                        let documents = documents.into_iter().enumerate().map(|(index, (path, document))| {
                            let actual = physical(&document);
                            let native = physical(&std::fs::read_to_string(directory.join(format!("native-{case}-{index}.xml"))).expect("complete native member"));
                            assert_eq!(actual, native, "complete native/generated physical member");
                            json!({"path":path,"document":document,"physical":actual,"native_physical_match":true})
                        }).collect::<Vec<_>>();
                        json!({"name":name,"documents":documents})
                    }).collect::<Vec<_>>();
                    println!("{}", json!({"case":case,"api":api,"kind":"ok","primary":{"document":primary,"physical":primary_physical,"native_physical_match":true},"extras":extras}));
                }
                Err(error) => {
                    assert!(std::ptr::eq(error.source().unwrap().downcast_ref::<XmlBoundaryError>().unwrap(), error.boundary.as_ref()), "standard source borrows actual boundary");
                    assert_eq!((error.boundary.bytes, error.boundary.limit), (None, None));
                    let (owner, runtime_error, node) = if matches!(case, "late-output" | "primary-output") {
                        assert_eq!(error.boundary.kind, XmlBoundaryErrorKind::Output);
                        let owner = if case == "primary-output" {
                            assert!(matches!(error.owner.as_ref(), Some(XmlDocumentOutputsOwner::Primary)));
                            json!({"kind":"Primary"})
                        } else {
                            let Some(XmlDocumentOutputsOwner::NamedMember { declaration_index, name, index, path }) = error.owner.as_ref() else { panic!("exact named member owner") };
                            assert_eq!((*declaration_index, name.as_str(), *index, path.as_str()), (0, "Audit", 1, "bad.xml"));
                            json!({"kind":"NamedMember","declaration_index":declaration_index,"name":name,"index":index,"path":path})
                        };
                        assert_eq!(boundary(&error.boundary), boundary(original_output.as_ref().unwrap()), "same-language original serializer boundary");
                        (owner, Json::Null, Json::Null)
                    } else {
                        assert!(case.starts_with("mapping-") && error.owner.is_none());
                        assert_eq!(error.boundary.kind, XmlBoundaryErrorKind::Mapping);
                        let cause = error.boundary.source().unwrap().downcast_ref::<RuntimeError>().expect("actual original mapping cause");
                        assert!(matches!(cause, RuntimeError::EmptyDynamicTargetPath { node: 0 }));
                        assert_eq!(cause.to_string(), *original_mapping.as_ref().unwrap());
                        (Json::Null, json!("EmptyDynamicTargetPath"), json!(0))
                    };
                    println!("{}", json!({"case":case,"api":api,"kind":format!("{:?}",error.boundary.kind),"owner":owner,"boundary":boundary(&error.boundary),"boundary_identity":true,"original_boundary_match":true,"runtime_error":runtime_error,"node":node,"no_partial_result":true}));
                }
            }
        }
    }
    Ok(())
}
