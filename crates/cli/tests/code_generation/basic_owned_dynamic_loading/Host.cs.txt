using System.Globalization;
using System.Text.Json;
using System.Text.Json.Nodes;
using Ferrule.Generated;
using Ferrule.Runtime;

/// Records every attempted callback; repeated successful documents are read
/// once and failures are never cached.
internal sealed class RecordingLoader(IReadOnlyDictionary<(string, string), FerruleInstance> documents) : IFerruleDynamicSourceLoader
{
    private readonly Dictionary<(string, string), FerruleInstance> _cache = new();
    public List<string> Calls { get; } = new();
    public List<string> Reads { get; } = new();
    public FerruleInstance Load(string sourceName, string logicalPath)
    {
        var key = (sourceName, logicalPath);
        Calls.Add($"{sourceName}:{logicalPath}");
        if (_cache.TryGetValue(key, out var cached)) return cached;
        Reads.Add($"{sourceName}:{logicalPath}");
        if (!documents.TryGetValue(key, out var document))
            throw new InvalidOperationException($"no document {sourceName}:{logicalPath}");
        _cache[key] = document;
        return document;
    }
}

internal static class Driver
{
    private static JsonNode? Json(object? value) => JsonSerializer.SerializeToNode(value);
    private static FerruleInstance Decode(JsonNode recipe)
    {
        var tag = recipe[0]!.GetValue<string>();
        if (tag == "Group")
            return new FerruleGroup(recipe[1]!.AsArray().Select(field => new FerruleField(field![0]!.GetValue<string>(), Decode(field[1]!))));
        if (tag == "Repeated") return new FerruleRepeated(recipe[1]!.AsArray().Select(item => Decode(item!)));
        return new FerruleScalar(tag switch {
            "Null" => FerruleValue.Null,
            "JsonNull" => FerruleValue.JsonNull,
            "String" => FerruleValue.FromString(recipe[1]!.GetValue<string>()),
            "Bool" => FerruleValue.FromBoolean(recipe[1]!.GetValue<bool>()),
            "Int" => FerruleValue.FromInt64(recipe[1]!.GetValue<long>()),
            _ => throw new InvalidOperationException("Closed authored scalar tag."),
        });
    }
    private static JsonNode Encode(FerruleInstance instance)
    {
        if (instance is FerruleGroup group) return new JsonArray(Json("Group"), Json(group.Fields.Select(field => new JsonArray(Json(field.Name), Encode(field.Value))).ToArray()),
            Json(new { kind = group.XmlTypeOrigin.Kind.ToString(), identity = group.XmlTypeOrigin.Identity, literal = group.XmlTypeOrigin.Literal }));
        if (instance is FerruleRepeated repeated) return new JsonArray(Json("Repeated"), Json(repeated.Items.Select(Encode).ToArray()));
        if (instance is not FerruleScalar scalar) throw new InvalidOperationException("Closed authored tree kind.");
        var value = scalar.Value;
        return value.Kind switch {
            FerruleValueKind.Null => new JsonArray(Json("Null")),
            FerruleValueKind.JsonNull => new JsonArray(Json("JsonNull")),
            FerruleValueKind.String => new JsonArray(Json("String"), Json(value.StringValue)),
            FerruleValueKind.Bool => new JsonArray(Json("Bool"), Json(value.BooleanValue)),
            FerruleValueKind.Int64 => new JsonArray(Json("Int"), Json(value.Int64Value)),
            _ => throw new InvalidOperationException("Closed typed output tag."),
        };
    }
    private static JsonNode Outputs(object output)
    {
        var result = new JsonArray();
        void Add(string? name, FerruleInstance instance) => result.Add(new JsonArray(Json(name), Encode(instance)));
        switch (output)
        {
            case ExecutionOutputs all: Add(null, all.Primary); foreach (var named in all.Extras) Add(named.Name, named.Instance); break;
            case SelectedTargetOutput.Primary primary: Add(null, primary.Instance); break;
            case SelectedTargetOutput.Named named: Add(named.Output.Name, named.Output.Instance); break;
            default: throw new InvalidOperationException("Unexpected public typed outcome.");
        }
        return result;
    }
    // Typed failure fields are compared; human wording is not a cross-language contract.
    private static bool ErrorMatches(Exception error, JsonNode recipe)
    {
        if (error is not FerruleRuntimeException runtime) return false;
        uint? node = recipe["node"] is JsonNode value ? value.GetValue<uint>() : null;
        return recipe["kind"]!.GetValue<string>() switch {
            "MissingLoader" => runtime.Error == FerruleRuntimeError.MissingDynamicSourceLoader
                && runtime.SourceField == recipe["source"]!.GetValue<string>(),
            "DynamicPath" => runtime.Error == FerruleRuntimeError.DynamicSourcePath
                && runtime.SourceField == recipe["source"]!.GetValue<string>()
                && runtime.Node == node
                && runtime.FoundKind == recipe["found"]!.GetValue<string>() switch {
                    "int" => FerruleValueKind.Int64,
                    _ => throw new InvalidOperationException("Closed authored type."),
                },
            "Load" => runtime.Error == FerruleRuntimeError.DynamicSourceLoad
                && runtime.SourceField == recipe["source"]!.GetValue<string>()
                && runtime.Detail == recipe["path"]!.GetValue<string>()
                && runtime.InnerException?.Message == recipe["message"]!.GetValue<string>(),
            // A loading scope's driver walk has no node; an owned read has its own.
            "OwnedSource" => runtime.Error == FerruleRuntimeError.OwnedSource
                && runtime.Node == node
                && runtime.OwnedSourceFailure is { } owned && ReferenceEquals(owned, runtime.InnerException)
                && owned.Owner?.Name == recipe["owner"]!.GetValue<string>()
                && owned.Error == recipe["cause"]!.GetValue<string>() switch {
                    "SourceUnavailable" => FerruleOwnedSourceErrorKind.SourceUnavailable,
                    _ => throw new InvalidOperationException("Closed authored owned cause."),
                },
            _ => throw new InvalidOperationException("Closed authored error recipe."),
        };
    }
    private static object Invoke(string selection, FerruleInstance source, NamedInput[] inputs,
        FerruleOwnedSourceLimits limits, FerruleExecutionContext context, RecordingLoader? loader)
    {
        if (selection == "All")
            return loader is null ? GeneratedMapping.ExecuteWithBasicOwnedSources(source, inputs, limits, context)
                : GeneratedMapping.ExecuteWithBasicOwnedSourcesAndDynamicSourceLoader(source, inputs, limits, context, loader);
        var target = selection == "Primary" ? FerruleTargetSelection.Primary() : FerruleTargetSelection.Named(selection);
        return loader is null ? GeneratedMapping.ExecuteSelectedWithBasicOwnedSources(source, target, inputs, limits, context)
            : GeneratedMapping.ExecuteSelectedWithBasicOwnedSourcesAndDynamicSourceLoader(source, target, inputs, limits, context, loader);
    }
    private static int Main(string[] args)
    {
        if (args.Length != 3) throw new InvalidOperationException("Controls, model ordinal and results path required.");
        var controls = JsonNode.Parse(File.ReadAllText(args[0]))!;
        var model = int.Parse(args[1], CultureInfo.InvariantCulture);
        var documents = controls["documents"]!.AsArray().ToDictionary(
            entry => (entry![0]!.GetValue<string>(), entry[1]!.GetValue<string>()), entry => Decode(entry![2]!));
        var limits = new FerruleOwnedSourceLimits(16, 32, 64, 16_777_216, 512);
        var context = new FerruleExecutionContext("dynload475.mapping", "dynload475.mapping", null);
        var rows = new JsonArray();
        foreach (var entry in controls["cases"]!.AsArray())
        {
            var item = entry!;
            if (item["model"]!.GetValue<int>() != model) continue;
            var source = Decode(item["source"]!);
            var inputs = item["inputs"]!.AsArray().Select(input => new NamedInput(input![0]!.GetValue<string>(), Decode(input[1]!))).ToArray();
            JsonNode Snapshot() => new JsonArray(Encode(source), Json(inputs.Select(input => Encode(input.Instance)).ToArray()));
            var before = Snapshot();
            foreach (var callNode in item["calls"]!.AsArray())
            {
                var call = callNode!;
                var selection = call["selection"]!.GetValue<string>();
                var recording = new RecordingLoader(documents);
                var loader = item["loader"]!.GetValue<bool>() ? recording : null;
                object actual;
                try { actual = Invoke(selection, source, inputs, limits, context, loader); }
                catch (Exception caught) { actual = caught; }
                var expected = call["expected"]!;
                var matched = (actual is Exception error
                    ? expected["Err"] is JsonNode recipe && ErrorMatches(error, recipe)
                    : expected["Ok"] is JsonNode trees && JsonNode.DeepEquals(Outputs(actual), trees))
                    && JsonNode.DeepEquals(Json(recording.Calls), call["calls"])
                    && JsonNode.DeepEquals(Json(recording.Reads), call["reads"])
                    && JsonNode.DeepEquals(before, Snapshot());
                rows.Add(Json(new { @case = item["id"]!.GetValue<string>(), selection, matched,
                    actual = actual is Exception failure ? failure.ToString() : Outputs(actual).ToJsonString(),
                    calls = recording.Calls, reads = recording.Reads }));
            }
        }
        File.WriteAllText(args[2], rows.ToJsonString());
        return rows.Count == @MODEL_CALLS@ && rows.All(row => row!["matched"]!.GetValue<bool>()) ? 0 : 1;
    }
}
