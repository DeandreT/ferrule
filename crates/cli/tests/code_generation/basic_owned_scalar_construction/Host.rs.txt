use codegen_runtime::{BasicNamedSourceInput, ExecutionContext, Instance, InstanceKind,
    OwnedSourceError, OwnedSourceIdentity, OwnedSourceLimitKind, OwnedSourceLimits, RuntimeError,
    SourcePathError, TargetSelection};
use serde_json::{Value as Json, json};
use std::path::Path;
#[allow(dead_code)]
mod codec;

fn identity(name: &str) -> OwnedSourceIdentity {
    if name == "Primary" { OwnedSourceIdentity::Primary } else { OwnedSourceIdentity::Named(name.into()) }
}
fn limit_kind(name: &str) -> OwnedSourceLimitKind {
    match name {
        "Roots" => OwnedSourceLimitKind::Roots,
        "Frames" => OwnedSourceLimitKind::Frames,
        "PathSegments" => OwnedSourceLimitKind::PathSegments,
        "MetadataBytes" => OwnedSourceLimitKind::MetadataBytes,
        "WalkContexts" => OwnedSourceLimitKind::WalkContexts,
        _ => panic!("authored limit kind"),
    }
}

fn expected_error(recipe: &Json) -> RuntimeError {
    let text = |key: &str| recipe[key].as_str().unwrap();
    let number = |key: &str| recipe[key].as_u64().unwrap();
    match text("kind") {
        "SourceUnavailable" => RuntimeError::OwnedSourceAtNode {
            node: number("node") as u32,
            source: OwnedSourceError::SourceUnavailable { name: text("owner").into() },
        },
        "Traversal" => {
            assert_eq!(text("found"), "scalar");
            RuntimeError::OwnedSourceAtNode {
                node: number("node") as u32,
                source: OwnedSourceError::Traversal {
                    owner: identity(text("owner")),
                    source: SourcePathError::CannotTraverse {
                        path: recipe["path"].as_array().unwrap().iter().map(|part| part.as_str().unwrap().to_owned()).collect(),
                        segment: number("segment") as usize,
                        found: InstanceKind::Scalar,
                    },
                },
            }
        }
        "NotABool" => {
            assert_eq!(text("found"), "string");
            RuntimeError::NotABool { node: number("node") as u32, found: "string" }
        }
        "MetadataLimit" => {
            let source = OwnedSourceError::MetadataLimit {
                kind: limit_kind(text("limit")),
                used: number("used") as usize,
                requested: number("requested") as usize,
                max: number("maximum") as usize,
            };
            match recipe["node"].as_u64() {
                Some(node) => RuntimeError::OwnedSourceAtNode { node: node as u32, source },
                None => RuntimeError::OwnedSource(source),
            }
        }
        "WalkLimit" => {
            let counts = &recipe["rust"];
            let count = |key: &str| counts[key].as_u64().unwrap() as usize;
            RuntimeError::OwnedSource(OwnedSourceError::MetadataLimit {
                kind: OwnedSourceLimitKind::WalkContexts,
                used: count("used"),
                requested: count("requested"),
                max: count("maximum"),
            })
        }
        _ => panic!("authored error recipe"),
    }
}

macro_rules! invoke {
    ($module:ident, $selection:expr, $source:expr, $inputs:expr, $execution:expr, $limits:expr) => {{
        if $selection == "All" {
            $module::execute_with_basic_owned_sources($source, $inputs, Some($execution), $limits)
                .map(|output| {
                    let mut trees = vec![(None, output.primary)];
                    trees.extend(output.extras.into_iter().map(|named| (Some(named.name), named.instance)));
                    trees
                })
        } else {
            let target = if $selection == "Primary" { TargetSelection::Primary } else { TargetSelection::Named($selection) };
            $module::execute_selected_with_basic_owned_sources($source, $inputs, Some($execution), $limits, target)
                .map(|output| match output {
                    $module::SelectedTargetOutput::Primary(instance) => vec![(None, instance)],
                    $module::SelectedTargetOutput::Named(named) => vec![(Some(named.name), named.instance)],
                })
        }
    }};
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let args = std::env::args_os().skip(1).collect::<Vec<_>>();
    assert_eq!(args.len(), 2, "controls and results paths");
    let controls: Json = serde_json::from_slice(&std::fs::read(&args[0])?)?;
    let execution = ExecutionContext::new(Path::new("scalar550.mapping"));
    let mut rows = Vec::new();
    for case in controls["cases"].as_array().unwrap() {
        let source = codec::decode(&case["source"]);
        let named = case["inputs"].as_array().unwrap().iter()
            .map(|input| (input[0].as_str().unwrap().to_owned(), codec::decode(&input[1])))
            .collect::<Vec<(String, Instance)>>();
        let inputs = named.iter().map(|(name, instance)| BasicNamedSourceInput { name, instance }).collect::<Vec<_>>();
        let before = (source.clone(), named.clone());
        for call in case["calls"].as_array().unwrap() {
            let selection = call["selection"].as_str().unwrap();
            let bound = |index: usize, default: usize| call["limits"].get(index).map_or(default, |value| value.as_u64().unwrap() as usize);
            let limits = OwnedSourceLimits { max_roots: bound(0, 16), max_active_frames: bound(1, 32),
                max_path_segments: bound(2, 200), max_metadata_bytes: bound(3, 16_777_216), max_walk_contexts: bound(4, 512) };
            let actual: Result<Vec<(Option<String>, Instance)>, RuntimeError> = match case["model"].as_u64().unwrap() {
@DISPATCH@
                _ => panic!("authored model"),
            };
            let expected = &call["expected"];
            let matched = match &actual {
                Ok(trees) => expected.get("Ok").is_some_and(|expected| json!(trees.iter()
                    .map(|(name, tree)| json!([name, codec::encode(tree)])).collect::<Vec<_>>()) == *expected),
                Err(error) => expected.get("Err").is_some() && *error == expected_error(&expected["Err"]),
            } && before == (source.clone(), named.clone());
            rows.push(json!({"case": case["id"], "selection": selection, "limits": call["limits"], "matched": matched,
                "actual": if matched { String::from("matched") } else { format!("{actual:?}") }}));
        }
    }
    std::fs::write(&args[1], serde_json::to_vec_pretty(&rows)?)?;
    assert_eq!(rows.len(), @CALLS@);
    if rows.iter().any(|row| row["matched"] != true) {
        std::process::exit(1);
    }
    Ok(())
}
