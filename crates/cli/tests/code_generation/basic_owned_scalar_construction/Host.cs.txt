using System.Globalization;
using System.Text.Json;
using System.Text.Json.Nodes;
using Ferrule.Generated;
using Ferrule.Runtime;

internal static class Driver
{
    private static JsonNode? Json(object? value) => JsonSerializer.SerializeToNode(value);
    private static FerruleInstance Decode(JsonNode recipe)
    {
        var tag = recipe[0]!.GetValue<string>();
        if (tag == "Group")
            return new FerruleGroup(recipe[1]!.AsArray().Select(field => new FerruleField(field![0]!.GetValue<string>(), Decode(field[1]!))));
        if (tag == "Repeated") return new FerruleRepeated(recipe[1]!.AsArray().Select(item => Decode(item!)));
        return new FerruleScalar(tag switch {
            "Null" => FerruleValue.Null,
            "String" => FerruleValue.FromString(recipe[1]!.GetValue<string>()),
            "Bool" => FerruleValue.FromBoolean(recipe[1]!.GetValue<bool>()),
            "Int" => FerruleValue.FromInt64(recipe[1]!.GetValue<long>()),
            "Float" => FerruleValue.FromDouble(BitConverter.Int64BitsToDouble(
                long.Parse(recipe[1]!.GetValue<string>(), NumberStyles.AllowHexSpecifier, CultureInfo.InvariantCulture))),
            _ => throw new InvalidOperationException("Closed authored scalar tag."),
        });
    }
    private static JsonNode Encode(FerruleInstance instance)
    {
        if (instance is FerruleGroup group) return new JsonArray(Json("Group"), Json(group.Fields.Select(field => new JsonArray(Json(field.Name), Encode(field.Value))).ToArray()),
            Json(new { kind = group.XmlTypeOrigin.Kind.ToString(), identity = group.XmlTypeOrigin.Identity, literal = group.XmlTypeOrigin.Literal }));
        if (instance is FerruleRepeated repeated) return new JsonArray(Json("Repeated"), Json(repeated.Items.Select(Encode).ToArray()));
        if (instance is not FerruleScalar scalar) throw new InvalidOperationException("Closed authored tree kind.");
        var value = scalar.Value;
        return value.Kind switch {
            FerruleValueKind.Null => new JsonArray(Json("Null")),
            FerruleValueKind.String => new JsonArray(Json("String"), Json(value.StringValue)),
            FerruleValueKind.Bool => new JsonArray(Json("Bool"), Json(value.BooleanValue)),
            FerruleValueKind.Int64 => new JsonArray(Json("Int"), Json(value.Int64Value)),
            FerruleValueKind.Double => new JsonArray(Json("Float"),
                Json(BitConverter.DoubleToInt64Bits(value.DoubleValue).ToString("x16", CultureInfo.InvariantCulture))),
            _ => throw new InvalidOperationException("Closed typed output tag."),
        };
    }
    private static JsonNode Outputs(object output)
    {
        var result = new JsonArray();
        void Add(string? name, FerruleInstance instance) => result.Add(new JsonArray(Json(name), Encode(instance)));
        switch (output)
        {
            case ExecutionOutputs all: Add(null, all.Primary); foreach (var named in all.Extras) Add(named.Name, named.Instance); break;
            case SelectedTargetOutput.Primary primary: Add(null, primary.Instance); break;
            case SelectedTargetOutput.Named named: Add(named.Output.Name, named.Output.Instance); break;
            default: throw new InvalidOperationException("Unexpected public typed outcome.");
        }
        return result;
    }
    private static FerruleOwnedSourceLimitKind LimitKind(string name) => name switch {
        "Roots" => FerruleOwnedSourceLimitKind.Roots,
        "Frames" => FerruleOwnedSourceLimitKind.Frames,
        "PathSegments" => FerruleOwnedSourceLimitKind.PathSegments,
        "MetadataBytes" => FerruleOwnedSourceLimitKind.MetadataBytes,
        "WalkContexts" => FerruleOwnedSourceLimitKind.WalkContexts,
        _ => throw new InvalidOperationException("Closed authored limit kind."),
    };
    // Typed failure fields are compared; human wording is not a cross-language contract.
    private static bool ErrorMatches(Exception error, JsonNode recipe)
    {
        if (error is not FerruleRuntimeException runtime) return false;
        string Text(string key) => recipe[key]!.GetValue<string>();
        nuint Number(string key) => (nuint)recipe[key]!.GetValue<ulong>();
        bool Owned(FerruleOwnedSourceErrorKind kind, out FerruleOwnedSourceException owned)
        {
            owned = runtime.OwnedSourceFailure!;
            return runtime.Error == FerruleRuntimeError.OwnedSource && owned is not null
                && ReferenceEquals(owned, runtime.InnerException) && owned.Error == kind;
        }
        bool OwnedAt(FerruleOwnedSourceErrorKind kind, out FerruleOwnedSourceException owned) =>
            Owned(kind, out owned) && runtime.Node == recipe["node"]!.GetValue<uint>();
        bool IsOwner(FerruleOwnedSourceIdentity? owner) =>
            Text("owner") == "Primary" ? owner is { IsPrimary: true } : owner?.Name == Text("owner");
        return Text("kind") switch {
            "SourceUnavailable" => OwnedAt(FerruleOwnedSourceErrorKind.SourceUnavailable, out var owned)
                && IsOwner(owned.Owner),
            "Traversal" => OwnedAt(FerruleOwnedSourceErrorKind.Traversal, out var owned) && IsOwner(owned.Owner)
                && owned.InnerException is FerruleOwnedSourceTraversalException cause
                && cause.Error == FerruleOwnedSourceTraversalKind.CannotTraverse
                && cause.Path.SequenceEqual(recipe["path"]!.AsArray().Select(part => part!.GetValue<string>()))
                && cause.Segment == Number("segment")
                && cause.Found == Text("found") switch {
                    "scalar" => FerruleOwnedSourceInstanceKind.Scalar,
                    _ => throw new InvalidOperationException("Closed authored found kind."),
                },
            "NotABool" => runtime.Error == FerruleRuntimeError.NotABool
                && runtime.Node == recipe["node"]!.GetValue<uint>()
                && runtime.FoundKind == Text("found") switch {
                    "string" => FerruleValueKind.String,
                    _ => throw new InvalidOperationException("Closed authored found kind."),
                },
            "MetadataLimit" => Owned(FerruleOwnedSourceErrorKind.MetadataLimit, out var owned)
                && runtime.Node == recipe["node"]?.GetValue<uint>()
                && owned.LimitKind == LimitKind(Text("limit")) && owned.Used == Number("used")
                && owned.Requested == Number("requested") && owned.Maximum == Number("maximum"),
            "WalkLimit" => runtime.Error == FerruleRuntimeError.BasicSourceBridge && runtime.Node is null
                && runtime.BasicSourceBridgeFailure is { } bridge && ReferenceEquals(bridge, runtime.InnerException)
                && bridge.Error == FerruleBasicSourceBridgeErrorKind.PairedMetadataLimit
                && bridge.Component == FerruleBasicSourceBridgeComponent.WalkContexts
                && bridge.Used == (nuint)recipe["csharp"]!["used"]!.GetValue<ulong>()
                && bridge.Requested == (nuint)recipe["csharp"]!["requested"]!.GetValue<ulong>()
                && bridge.Maximum == (nuint)recipe["csharp"]!["maximum"]!.GetValue<ulong>(),
            _ => throw new InvalidOperationException("Closed authored error recipe."),
        };
    }
    private static string Describe(Exception failure)
    {
        if (failure is not FerruleRuntimeException runtime) return failure.ToString();
        var owned = runtime.OwnedSourceFailure;
        var traversal = owned?.InnerException as FerruleOwnedSourceTraversalException;
        return JsonSerializer.Serialize(new {
            error = runtime.Error.ToString(), node = runtime.Node, found = runtime.FoundKind?.ToString(),
            owned = owned?.Error.ToString(), owner = owned?.Owner?.Name, primary = owned?.Owner?.IsPrimary,
            requested = owned?.RequestedName, limit = owned?.LimitKind?.ToString(),
            used = owned?.Used?.ToString(), requestedCount = owned?.Requested?.ToString(), maximum = owned?.Maximum?.ToString(),
            traversal = traversal?.Error.ToString(), path = traversal?.Path, segment = traversal?.Segment?.ToString(),
            instance = traversal?.Found.ToString(), bridge = runtime.BasicSourceBridgeFailure?.Error.ToString(),
            component = runtime.BasicSourceBridgeFailure?.Component?.ToString(),
            bridgeUsed = runtime.BasicSourceBridgeFailure?.Used?.ToString(),
            bridgeRequested = runtime.BasicSourceBridgeFailure?.Requested?.ToString(),
            bridgeMaximum = runtime.BasicSourceBridgeFailure?.Maximum?.ToString(),
            message = runtime.Message,
        });
    }
    private static int Main(string[] args)
    {
        if (args.Length != 3) throw new InvalidOperationException("Controls, model ordinal and results path required.");
        var controls = JsonNode.Parse(File.ReadAllText(args[0]))!;
        var model = int.Parse(args[1], CultureInfo.InvariantCulture);
        var context = new FerruleExecutionContext("scalar550.mapping", "scalar550.mapping", null);
        var rows = new JsonArray();
        foreach (var entry in controls["cases"]!.AsArray())
        {
            var item = entry!;
            if (item["model"]!.GetValue<int>() != model) continue;
            var source = Decode(item["source"]!);
            var inputs = item["inputs"]!.AsArray().Select(input => new NamedInput(input![0]!.GetValue<string>(), Decode(input[1]!))).ToArray();
            JsonNode Snapshot() => new JsonArray(Encode(source), Json(inputs.Select(input => Encode(input.Instance)).ToArray()));
            var before = Snapshot();
            foreach (var callNode in item["calls"]!.AsArray())
            {
                var call = callNode!;
                var selection = call["selection"]!.GetValue<string>();
                nuint Bound(int index, nuint fallback) => call["limits"] is JsonArray bounds ? (nuint)bounds[index]!.GetValue<ulong>() : fallback;
                var limits = new FerruleOwnedSourceLimits(Bound(0, 16), Bound(1, 32), Bound(2, 200), Bound(3, 16_777_216), Bound(4, 512));
                object actual;
                try
                {
                    actual = selection == "All" ? GeneratedMapping.ExecuteWithBasicOwnedSources(source, inputs, limits, context)
                        : GeneratedMapping.ExecuteSelectedWithBasicOwnedSources(source,
                            selection == "Primary" ? FerruleTargetSelection.Primary() : FerruleTargetSelection.Named(selection), inputs, limits, context);
                }
                catch (Exception caught) { actual = caught; }
                var expected = call["expected"]!;
                var matched = (actual is Exception error
                    ? expected["Err"] is JsonNode recipe && ErrorMatches(error, recipe)
                    : expected["Ok"] is JsonNode trees && JsonNode.DeepEquals(Outputs(actual), trees))
                    && JsonNode.DeepEquals(before, Snapshot());
                string Report()
                {
                    if (actual is Exception failure) return Describe(failure);
                    try { return Outputs(actual).ToJsonString(); }
                    catch (Exception unreadable) { return unreadable.GetType().Name; }
                }
                rows.Add(Json(new { @case = item["id"]!.GetValue<string>(), selection, matched,
                    actual = matched ? "matched" : Report() }));
            }
        }
        File.WriteAllText(args[2], rows.ToJsonString());
        return rows.Count == @MODEL_CALLS@ && rows.All(row => row!["matched"]!.GetValue<bool>()) ? 0 : 1;
    }
}
