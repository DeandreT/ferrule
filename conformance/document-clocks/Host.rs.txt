use std::{cell::Cell, ops::ControlFlow, path::Path};
use codegen_runtime::{DocumentMember, ExecutionContext, Instance, RuntimeError, Value, group, repeated};
use document_clock_mapping::{DocumentVisit, visit_target_documents, visit_target_documents_with_filename_clock as visit};

fn source(flags: &[(bool, i64)]) -> Instance {
    Instance::DocumentSet(flags.iter().enumerate().map(|(index,(fail,mode))| DocumentMember::new(format!("{index}.xml"), group([
        ("Fail".into(),Instance::Scalar(Value::Bool(*fail))),
        ("Mode".into(),Instance::Scalar(Value::Int(*mode))),
        ("Samples".into(),repeated((0..2).map(|_|group([("Marker".into(),Instance::Scalar(Value::String("x".into())))])))),
    ])).unwrap()).collect())
}
fn main() {
    let execution=ExecutionContext::new(Path::new("clock.ferrule")).with_current_datetime("body-clock");
    let calls=Cell::new(0);
    let clock=|| { calls.set(calls.get()+1); Ok(format!("{}-",calls.get())) };
    let input=source(&[(false,1),(false,0),(false,2),(false,3)]);
    let mut documents=Vec::new();
    let outcome=visit(&input,&[],Some(&execution),None,&clock,|document| { documents.push(document); ControlFlow::<()>::Continue(()) }).unwrap();
    assert_eq!(outcome,DocumentVisit::Complete {accepted:4});
    assert_eq!(calls.get(),5);
    assert_eq!(documents.iter().map(|document| document.path()).collect::<Vec<_>>(),["1-0.xml","fixed-1.xml","2-3-2.xml","5-3.xml"]);
    assert!(documents.iter().all(|document|document.value().field("Body").and_then(Instance::as_scalar)==Some(&Value::String("body-clock".into()))));
    let mut stable=Vec::new();
    let outcome=visit_target_documents(&input,&[],Some(&execution),None,|document| { stable.push(document);ControlFlow::<()>::Continue(()) }).unwrap();
    assert_eq!(outcome,DocumentVisit::Complete {accepted:4});
    assert_eq!(calls.get(),5);
    assert_eq!(Instance::DocumentSet(stable),document_clock_mapping::execute_with_context(&input,&execution).unwrap());
    let before=calls.get();
    let failure=visit::<()>(&source(&[(true,1)]),&[],Some(&execution),None,&clock,|_|panic!("failed body offered")).unwrap_err();
    assert_eq!(failure.accepted,0);
    assert!(matches!(failure.source,RuntimeError::MappingException {node:2,..}));
    assert_eq!(calls.get(),before);
    let outcome=visit(&source(&[(false,1),(true,1)]),&[],Some(&execution),None,&clock,|_|ControlFlow::Break("stop")).unwrap();
    assert_eq!(outcome,DocumentVisit::Stopped {accepted:0,reason:"stop"});
    assert_eq!(calls.get(),before+1);
    let attempts=Cell::new(0);
    let failing=|| {attempts.set(attempts.get()+1);if attempts.get()==2 {Err("clock unavailable".into())}else {Ok("good-".into())}};
    let mut kept=Vec::new();
    let failure=visit(&source(&[(false,1);3]),&[],Some(&execution),None,&failing,|document|{kept.push(document);ControlFlow::<()>::Continue(())}).unwrap_err();
    assert_eq!(failure.accepted,1);
    assert!(matches!(failure.source,RuntimeError::DocumentFilenameClock {node:Some(0),ref message} if message=="clock unavailable"));
    assert_eq!(attempts.get(),2);
    assert_eq!(kept[0].path(),"good-0.xml");
    let outcome=visit::<()>(&source(&[]),&[],Some(&execution),None,&clock,|_|panic!("empty source offered")).unwrap();
    assert_eq!(outcome,DocumentVisit::Complete {accepted:0});
    println!("PASS document filename clocks");
}
