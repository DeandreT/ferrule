use std::{cell::{Cell,RefCell}, ops::ControlFlow, path::Path};
use codegen_runtime::{DocumentMember, ExecutionContext, Instance, RuntimeError, Value, group, repeated};
use document_body_clock_mapping::{DocumentVisit, visit_target_documents, visit_target_documents_with_clocks as visit};

fn source(flags: &[(bool, i64)]) -> Instance {
    Instance::DocumentSet(flags.iter().enumerate().map(|(index,(fail,mode))| DocumentMember::new(format!("{index}.xml"), group([
        ("Fail".into(),Instance::Scalar(Value::Bool(*fail))),
        ("Mode".into(),Instance::Scalar(Value::Int(*mode))),
        ("Samples".into(),repeated((0..2).map(|_|group([("Marker".into(),Instance::Scalar(Value::String("x".into())))])))),
    ])).unwrap()).collect())
}
fn text(document:&DocumentMember,name:&str)->String {
    let Some(Value::String(value))=document.value().field(name).and_then(Instance::as_scalar) else {panic!("missing field {name}")};value.clone()
}
fn main() {
    let execution=ExecutionContext::new(Path::new("clock.ferrule")).with_current_datetime("stable").with_invocation_counter_limits(codegen_runtime::InvocationCounterLimits{max_counters:1,max_text_bytes:64});
    let body_calls=Cell::new(0);let file_calls=Cell::new(0);let events=RefCell::new(Vec::new());
    let body=||{body_calls.set(body_calls.get()+1);let value=format!("B{}",body_calls.get());events.borrow_mut().push(value.clone());Ok(value)};
    let file=||{file_calls.set(file_calls.get()+1);let value=format!("F{}",file_calls.get());events.borrow_mut().push(value.clone());Ok(value+"-")};
    let mut documents=Vec::new();
    let input=source(&[(false,1),(false,0),(false,2),(false,3)]);
    let result=visit(&input,&[],Some(&execution),None,&body,&file,|document|{documents.push(document);ControlFlow::<()>::Continue(())}).unwrap();
    assert_eq!(result,DocumentVisit::Complete{accepted:4});assert_eq!((body_calls.get(),file_calls.get()),(16,5));
    assert_eq!(documents.iter().map(|d|d.path()).collect::<Vec<_>>(),["F1-0.xml","fixed-1.xml","F2-F3-2.xml","F5-3.xml"]);
    assert_eq!(events.borrow().iter().map(String::as_str).collect::<Vec<_>>(),["B1","B2","B3","B4","F1","B5","B6","B7","B8","B9","B10","B11","B12","F2","F3","B13","B14","B15","B16","F4","F5"]);
    for (index,document) in documents.iter().enumerate(){assert_eq!(text(document,"Body"),format!("B{}",index*4+1));assert_eq!(text(document,"Repeated"),format!("B{}",index*4+2));assert_eq!(text(document,"Fold"),format!("B{}",index*4+4));assert_eq!(document.value().field("Count").and_then(Instance::as_scalar),Some(&Value::Int(index as i64+1)));}
    let mut stable=Vec::new();let stable_result=visit_target_documents(&input,&[],Some(&execution),None,|d|{stable.push(d);ControlFlow::<()>::Continue(())}).unwrap();assert_eq!(stable_result,DocumentVisit::Complete{accepted:4});
    assert!(stable.iter().all(|d|["Body","Repeated","Fold"].iter().all(|field|text(d,field)=="stable")));
    assert!(stable.iter().enumerate().all(|(index,d)|d.value().field("Count").and_then(Instance::as_scalar)==Some(&Value::Int(index as i64+1))));
    assert_eq!((body_calls.get(),file_calls.get()),(16,5));
    let attempts=Cell::new(0);let files=Cell::new(0);
    let fail_body=||{attempts.set(attempts.get()+1);if attempts.get()==5{Err("body unavailable".into())}else{Ok("body".into())}};
    let filename=||{files.set(files.get()+1);Ok("file-".into())};
    let failure=visit::<()>(&source(&[(false,1);3]),&[],Some(&execution),None,&fail_body,&filename,|_|ControlFlow::Continue(())).unwrap_err();
    assert_eq!(failure.accepted,1);assert!(matches!(failure.source,RuntimeError::DocumentBodyClock{node:Some(0),ref message} if message=="body unavailable"));assert_eq!((attempts.get(),files.get()),(5,1));
    let body_count=Cell::new(0);let name_count=Cell::new(0);
    let body=||{body_count.set(body_count.get()+1);Ok("body".into())};
    let name=||{name_count.set(name_count.get()+1);if name_count.get()==2{Err("filename unavailable".into())}else{Ok("file-".into())}};
    let failure=visit::<()>(&source(&[(false,1);3]),&[],Some(&execution),None,&body,&name,|_|ControlFlow::Continue(())).unwrap_err();
    assert_eq!(failure.accepted,1);assert!(matches!(failure.source,RuntimeError::DocumentFilenameClock{node:Some(0),..}));assert_eq!((body_count.get(),name_count.get()),(8,2));
    let before=(body_calls.get(),file_calls.get());
    let stopped=visit(&source(&[(false,1),(true,1)]),&[],Some(&execution),None,&body,&file,|_|ControlFlow::Break("stop")).unwrap();
    assert_eq!(stopped,DocumentVisit::Stopped{accepted:0,reason:"stop"});assert_eq!(body_count.get(),12);assert_eq!(file_calls.get(),before.1+1);
    let empty=visit::<()>(&source(&[]),&[],Some(&execution),None,&body,&file,|_|panic!("empty document offered")).unwrap();assert_eq!(empty,DocumentVisit::Complete{accepted:0});
    let before=(body_count.get(),file_calls.get());
    assert!(visit::<()>(&source(&[(true,1)]),&[],Some(&execution),None,&body,&file,|_|panic!("failed body offered")).is_err());assert_eq!((body_count.get(),file_calls.get()),before);
    println!("PASS document body clocks");
}
