using Ferrule.Generated;
using Ferrule.Runtime;

static FerruleInstance Source(params (string Order, FerruleValue? Item)[] rows) =>
    new FerruleGroup(new[] {new FerruleField("Rows", new FerruleRepeated(rows.Select(row => {
        var fields = new List<FerruleField> {new("Order", new FerruleScalar(FerruleValue.FromString(row.Order)))};
        if (row.Item is not null) fields.Add(new("Item", new FerruleScalar(row.Item.Value)));
        return new FerruleGroup(fields);
    })))});
static void Check(FerruleDocument document, string path, params (string Item, long Line)[] expected) {
    if (document.Path != path || document.Value is not FerruleRepeated rows || rows.Items.Count != expected.Length)
        throw new InvalidOperationException("Document membership differs.");
    for (int i=0; i<expected.Length; i++) {
        if (rows.Items[i] is not FerruleGroup row || row.Fields.Count != 2 ||
            row.Fields[0].Name != "Item" || ((FerruleScalar)row.Fields[0].Value).Value.StringValue != expected[i].Item ||
            row.Fields[1].Name != "Line" || ((FerruleScalar)row.Fields[1].Value).Value.Int64Value != expected[i].Line)
            throw new InvalidOperationException("Complete row value differs.");
    }
}
